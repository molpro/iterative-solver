66 void check_hermitian(const CVecRef<R>& params, const CVecRef<R>& metric_actions,
67 const subspace::IXSpace<R, Q, P>& xspace, ArrayHandlers<R, Q, P>& handlers) const {
68 const Q* q = nullptr;
69 const Q* metric_action_q = nullptr;
70 value_type qmq = 0; // <q|M q>
71 if (xspace.stores_metric_actions()) {
72 if (xspace.dimensions().nQ == 0)
73 return;
74 q = &xspace.cparamsq().front().get();
75 metric_action_q = &xspace.cmetric_actionsq().front().get();
76 qmq = xspace.data.at(subspace::EqnData::M)(xspace.dimensions().oQ(), xspace.dimensions().oQ());
77 } else if (m_reference) {
78 q = &m_reference->q;
79 metric_action_q = &m_reference->metric_action_q;
80 qmq = m_reference->qmq;
81 } else
82 return;
83 // <x|M q> and <M x|q> from one pass over q and M q
84 auto left = params;
85 left.insert(left.end(), metric_actions.begin(), metric_actions.end());
86 const auto overlaps =
87 subspace::util::overlap(left, CVecRef<Q>{std::cref(*metric_action_q), std::cref(*q)}, handlers.rq());
88 for (size_t i = 0; i < params.size(); ++i) {
89 const auto xmq = overlaps(i, 0);
90 const auto mxq = overlaps(params.size() + i, 1);
91 // the size of these elements, which the metric bounds with its values on x and q
92 const auto scale = std::sqrt(std::abs(handlers.rr().dot(params[i], metric_actions[i]) * qmq));
93 if (std::abs(xmq - mxq) > hermiticity_tolerance() * scale)
94 throw std::runtime_error(std::format("The metric is not Hermitian: <x|M q> = {:.6e} differs from <M x|q> = "
95 "{:.6e} for a new parameter x and a subspace vector q",
96 double(std::abs(xmq)), double(std::abs(mxq))));
97 }
98 }

121 static void check_hermitian(const array::Span<value_type>& matrix, const size_t n) {
122 // the diagonal elements, which have to be real, are compared with themselves
123 for (size_t i = 0; i < n; ++i)
124 for (size_t j = 0; j <= i; ++j)
125 if (std::abs(matrix[i * n + j] - conjugate(matrix[j * n + i])) >
126 hermiticity_tolerance() * std::sqrt(std::abs(matrix[i * n + i] * matrix[j * n + j])))
127 throw std::runtime_error("The metric is not Hermitian: its matrix in the P space is not");
128 }

154 void check_positive_subspace(const subspace::Matrix<value_type>& metric, const subspace::Matrix<value_type>& overlap,
155 const size_t nP) {
156 using matrix_type = Eigen::Matrix<value_type, Eigen::Dynamic, Eigen::Dynamic>;
157 using row_major_type = Eigen::Matrix<value_type, Eigen::Dynamic, Eigen::Dynamic, Eigen::RowMajor>;
158 const auto n = Eigen::Index(metric.rows());
159 const auto np = Eigen::Index(nP);
160 if (np == 0 || np >= n)
161 return check_positive(metric, overlap);
162 const matrix_type m_rows = Eigen::Map<const row_major_type>(metric.data().data(), n, n);
163 const matrix_type m = (m_rows + m_rows.adjoint()) / value_type(2);
164 if (!m_pp_factor || m_pp_factor->rows() != np) {
165 const Eigen::LLT<matrix_type> llt(m.topLeftCorner(np, np));
166 if (llt.info() != Eigen::Success)
167 return check_positive(metric, overlap);
168 m_pp_factor = matrix_type(llt.matrixL());
169 }
170 const matrix_type y = m_pp_factor->template triangularView<Eigen::Lower>().solve(m.topRightCorner(np, n - np));
171 const matrix_type schur = m.bottomRightCorner(n - np, n - np) - y.adjoint() * y;
172 if (Eigen::LLT<matrix_type>(schur).info() == Eigen::Success)
173 return;
174 check_positive(metric, overlap);
175 }

177 static void check_positive(const subspace::Matrix<value_type>& metric, const subspace::Matrix<value_type>& overlap) {
178 using matrix_type = Eigen::Matrix<value_type, Eigen::Dynamic, Eigen::Dynamic>;
179 using row_major_type = Eigen::Matrix<value_type, Eigen::Dynamic, Eigen::Dynamic, Eigen::RowMajor>;
180 const auto n = Eigen::Index(metric.rows());
181 if (n == 0)
182 return;
183 const matrix_type m_rows = Eigen::Map<const row_major_type>(metric.data().data(), n, n);
184 const matrix_type m = (m_rows + m_rows.adjoint()) / value_type(2);
185 if (Eigen::LLT<matrix_type>(m).info() == Eigen::Success)
186 return;
187 const matrix_type s = Eigen::Map<const row_major_type>(overlap.data().data(), n, n);
188 const Eigen::SelfAdjointEigenSolver<matrix_type> s_eigen(s);
189 const value_type_abs s_threshold = precision_scaled<value_type>(1e-10) * s_eigen.eigenvalues().maxCoeff();
190 Eigen::Index rank = 0;
191 for (Eigen::Index i = 0; i < n; ++i)
192 if (s_eigen.eigenvalues()(i) > s_threshold)
193 ++rank;
194 if (rank == 0)
195 return;
196 // the eigenvalues come in ascending order, so the retained directions are the trailing ones
197 const auto retained = s_eigen.eigenvectors().rightCols(rank);
198 const matrix_type orthonormal =
199 retained * s_eigen.eigenvalues().tail(rank).cwiseSqrt().cwiseInverse().template cast<value_type>().asDiagonal();
200 const Eigen::SelfAdjointEigenSolver<matrix_type> m_eigen(orthonormal.adjoint() * m * orthonormal,
201 Eigen::EigenvaluesOnly);
202 const value_type_abs lowest = m_eigen.eigenvalues().minCoeff();
203 const value_type_abs largest = m_eigen.eigenvalues().cwiseAbs().maxCoeff();
204 // the rounding errors of the elements of the metric matrix, amplified by the orthonormalisation
205 const value_type_abs rounding = precision_scaled<value_type>(1e-13) * m.norm() / s_eigen.eigenvalues()(n - rank);
206 if (lowest < -positivity_tolerance() * largest - rounding)
207 throw std::runtime_error(std::format("The metric is not positive definite: it has the Rayleigh quotient {:.3e} "
208 "on the subspace, against a largest one of {:.3e}",
209 double(lowest), double(largest)));
210 }
