154 void solution_with_errors(const std::vector<int>& roots, const VecRef<R>& parameters, const VecRef<R>& residual,
155 const VecRef<R>& scratch, const std::vector<value_type_abs>& solution_norms,
156 std::vector<value_type_abs>& errors) override {
157 auto truncated_partner_residual_norm = std::optional<value_type_abs>{};
158 this->solution_impl(roots, parameters, residual, scratch, [&](const auto&... args) {
159 construct_residual(args..., &truncated_partner_residual_norm);
160 });
161 SolverTemplate::compute_errors(roots, cwrap(parameters), cwrap(residual), solution_norms, errors);
162 if (!truncated_partner_residual_norm)
163 return;
164 auto subspace_solver = std::dynamic_pointer_cast<subspace::SubspaceSolverLinEig<R, Q, P>>(this->m_subspace_solver);
165 assert(subspace_solver);
166 const auto nroots = subspace_solver->size();
167 auto partner_error = *truncated_partner_residual_norm;
168 if (!solution_norms.empty())
169 // as in SolverTemplate::compute_errors(), the error of the partner of unit length
170 partner_error /= subspace::util::solution_norms(subspace_solver->retained_solutions(),
171 this->m_xspace->data.at(subspace::EqnData::S))
172 [subspace_solver->complex_partner(nroots - 1).value()];
173 auto& error = errors.at(std::ranges::distance(roots.begin(), std::ranges::find(roots, int(nroots) - 1)));
174 error = std::sqrt(error * error + partner_error * partner_error);
175 }

182 void construct_residual(const std::vector<int>& roots, const CVecRef<R>& params, const VecRef<R>& actions,
183 const detail::MetricSolutions<R, Q>& metric_solutions,
184 std::optional<value_type_abs>* truncated_partner_residual_norm) {
185 if (roots.empty())
186 return;
187 auto prof = this->profiler()->push("itsolv::construct_residual");
188 assert(params.size() >= roots.size());
189 const auto& eigvals = eigenvalues();
190
191 auto subspace_solver = std::dynamic_pointer_cast<subspace::SubspaceSolverLinEig<R, Q, P>>(this->m_subspace_solver);
192 assert(subspace_solver);
193 const auto& imag_eigval_components = subspace_solver->imag_eigval_components();
194
195 // With a metric M, the residual contains M x rather than x for each solution x. M x is given in metric_solutions
196 // where the metric is applied on the fly, and otherwise assembled from the stored actions of the metric on the Q
197 // and D spaces and the action of the metric on the P space. The coefficients are collected in these matrices.
198 const auto& xspace = *this->m_xspace;
199 const bool metric = xspace.metric_kind() != MetricKind::Identity;
200 const bool on_the_fly = metric && xspace.metric_storage() == MetricStorage::OnTheFly;
201 const bool stored = metric && !on_the_fly;
202 const auto& dims = xspace.dimensions();
203 const auto& solutions = this->m_subspace_solver->solutions();
204 using coefficient_matrix = std::decay_t<decltype(solutions)>;
205 // the coefficients of the Q and D vectors are in one matrix, with the Q vectors first
206 coefficient_matrix mp_coeff({stored ? dims.nP : 0, roots.size()}),
207 mqd_coeff({stored ? dims.nQ + dims.nD : 0, roots.size()}),
208 mx_coeff({on_the_fly ? roots.size() : 0, roots.size()});
209 // adds weight times the action of the metric on the solution params[k], whose coefficients are in the given row of
210 // coefficients, to the residual actions[i]
211 auto add_metric_solution = [&](const scalar_type weight, const coefficient_matrix& coefficients, const int row,
212 const size_t k, const size_t i) {
213 if (on_the_fly) {
214 mx_coeff(k, i) += weight;
215 } else {
216 for (size_t j = 0; j < dims.nP; ++j)
217 mp_coeff(j, i) += weight * coefficients(row, dims.oP() + j);
218 for (size_t j = 0; j < dims.nQ; ++j)
219 mqd_coeff(j, i) += weight * coefficients(row, dims.oQ() + j);
220 for (size_t j = 0; j < dims.nD; ++j)
221 mqd_coeff(dims.nQ + j, i) += weight * coefficients(row, dims.oD() + j);
222 }
223 };
224
225 for (size_t i = 0; i < roots.size(); ++i) {
226 if (metric)
227 add_metric_solution(-eigvals.at(roots[i]), solutions, roots[i], i, i);
228 else
229 this->m_handlers->rr().axpy(-eigvals.at(roots[i]), params.at(i), actions.at(i));
230
231 auto it = std::ranges::find(imag_eigval_components, roots[i], [](const auto& pair) { return pair.first; });
232
233 if (it != imag_eigval_components.end()) {
234 // Ref.: https://doi.org/10.1063/1.2755681 (Appendix)
235 // The idea is the following: We make use of the fact that the real and imaginary parts of the
236 // eigenvectors belonging to the complex root pair are spanned by the same basis. This allows
237 // us to split the associated residual vectors into a part corresponding to the real and a part
238 // corresponding to the imaginary part as well. The actual (complex-valued) residual can then
239 // always be reconstructed from this and therefore we don't lose any information. The formulas are
240 // |(R_real)_i> = sum_j [ (|A_j> - (e_real)_i |b_j>) (c_real)_{ji} + (e_imag) |b_j> (c_imag)_{ji} ]
241 // |(R_imag)_i> = sum_j [ (|A_j> - (e_real)_i |b_j>) (c_imag)_{ji} - (e_imag) |b_j> (c_real)_{ji} ]
242 // where |A_j> is the j-th action and |b_j> the j-th trial vector. e are the complex conjugate
243 // eigenvalues and c the associated eigenvectors.
244 // Note: At this point, params and actions already contain the vectors transformed into the
245 // eigenbasis (aka.: multiplied with c). Due to the way we process these complex-valued vectors,
246 // the first vector in a pair is the one that has been transformed with the real and the second
247 // with the imaginary part of c.
248 // The above code has already dealt with the bulk of the necessary expression and we only need
249 // to add the part with the imaginary components of the eigenvalue pair, for which the vector of
250 // the partner root is used.
251 // Note: The minus sign in above expression is implicitly taken into account as the
252 // imaginary parts of the complex conjugate eigenvalue pair has flipped signs
253 const auto partner = subspace_solver->complex_partner(roots[i]).value();
254 const auto root_it = std::ranges::find(roots, int(partner));
255 // the partner's vector, or the action of the metric on it applied on the fly, if it is assembled
256 auto partner_param = std::optional<R>{};
257 if (root_it != roots.end() && (!metric || on_the_fly)) {
258 const auto k = std::ranges::distance(roots.begin(), root_it);
259 if (metric)
260 add_metric_solution(it->second, solutions, int(partner), k, i);
261 else
262 this->m_handlers->rr().axpy(it->second, params.at(k), actions.at(i));
263 } else if (stored) {
264 // The partner's coefficients are among the retained solutions also if it is not among roots
265 if (partner >= subspace_solver->retained_solutions().rows())
266 throw std::logic_error("The complex conjugate partner of a root is not among the retained solutions");
267 add_metric_solution(it->second, subspace_solver->retained_solutions(), int(partner), 0, i);
268 } else {
269 // The partner is not among roots, or was truncated away, so its vector is assembled from the subspace
270 if (partner >= subspace_solver->retained_solutions().rows())
271 throw std::logic_error("The complex conjugate partner of a root is not among the retained solutions");
272 partner_param.emplace(this->m_handlers->rr().copy(actions.at(i)));
273 this->construct_solutions({int(partner)}, subspace_solver->retained_solutions(), wrap_arg(*partner_param));
274 if (on_the_fly)
275 apply_metric(*partner_param);
276 this->m_handlers->rr().axpy(it->second, *partner_param, actions.at(i));
277 }
278 if (partner >= eigvals.size() && truncated_partner_residual_norm) {
279 // The residual of the truncated partner p is the imaginary part of the residual of this root x. It is
280 // A p - e M p - f M x, where e and f are the real and imaginary parts of the eigenvalue of x, the imaginary
281 // part of the eigenvalue of p being -f, and M is the identity without a metric.
282 const auto& retained = subspace_solver->retained_solutions();
283 auto partner_residual = this->m_handlers->rr().copy(actions.at(i));
284 this->construct_actions({int(partner)}, retained, wrap_arg(partner_residual));
285 if (stored) {
286 auto metric_action = this->m_handlers->rr().copy(actions.at(i));
287 construct_subspace_metric_action(retained, int(partner), metric_action);
288 this->m_handlers->rr().axpy(-eigvals.at(roots[i]), metric_action, partner_residual);
289 construct_subspace_metric_action(retained, roots[i], metric_action);
290 this->m_handlers->rr().axpy(-it->second, metric_action, partner_residual);
291 } else {
292 this->m_handlers->rr().axpy(-eigvals.at(roots[i]), *partner_param, partner_residual);
293 if (!metric) {
294 this->m_handlers->rr().axpy(-it->second, params.at(i), partner_residual);
295 } else if (!metric_solutions.r.empty()) {
296 this->m_handlers->rr().axpy(-it->second, metric_solutions.r.at(i), partner_residual);
297 } else {
298 assert(metric_solutions.q.size() > i);
299 this->m_handlers->rq().axpy(-it->second, metric_solutions.q.at(i), partner_residual);
300 }
301 }
302 *truncated_partner_residual_norm = util::norm(this->m_handlers->rr(), partner_residual);
303 }
304 }
305 }
306 const auto residuals = wrap(actions.begin(), actions.begin() + roots.size());
307 if (stored) {
308 if (dims.nP > 0) {
309 using VectorP = typename SolverTemplate::VectorP;
310 auto pvectors = std::vector<VectorP>(roots.size(), VectorP(dims.nP));
311 for (size_t i = 0; i < roots.size(); ++i)
312 for (size_t j = 0; j < dims.nP; ++j)
313 pvectors[i][j] = mp_coeff(j, i);
314 this->m_apply_metric_p(pvectors, xspace.cparamsp(), residuals);
315 }
316 // one pass over the residuals for the actions of the metric on the Q and D vectors
317 auto qd_metric_actions = xspace.cmetric_actionsq();
318 const auto d_metric_actions = xspace.cmetric_actionsd();
319 qd_metric_actions.insert(qd_metric_actions.end(), d_metric_actions.begin(), d_metric_actions.end());
320 this->m_handlers->rq().gemm_outer(mqd_coeff, qd_metric_actions, residuals);
321 } else if (on_the_fly && !metric_solutions.r.empty()) {
322 this->m_handlers->rr().gemm_outer(mx_coeff, metric_solutions.r, residuals);
323 } else if (on_the_fly) {
324 this->m_handlers->rq().gemm_outer(mx_coeff, metric_solutions.q, residuals);
325 }
326 }

334 void combine_errors() override {
335 auto subspace_solver = std::dynamic_pointer_cast<subspace::SubspaceSolverLinEig<R, Q, P>>(this->m_subspace_solver);
336 assert(subspace_solver);
337 const auto& errors = subspace_solver->errors();
338 for (size_t root = 0; root < subspace_solver->size(); ++root) {
339 const auto partner = subspace_solver->complex_partner(root);
340 if (!partner || *partner <= root || *partner >= subspace_solver->size())
341 continue;
342 const auto error = std::sqrt(errors[root] * errors[root] + errors[*partner] * errors[*partner]);
343 subspace_solver->set_error(int(root), error);
344 subspace_solver->set_error(int(*partner), error);
345 }
346 }

356 void construct_subspace_metric_action(const subspace::Matrix<scalar_type>& solutions, int row, R& metric_action) {
357 const auto& dims = this->m_xspace->dimensions();
358 assert(solutions.cols() == dims.nX());
359 util::construct_solutions(wrap_arg(metric_action), {row}, solutions, CVecRef<P>{},
360 this->m_xspace->cmetric_actionsq(), this->m_xspace->cmetric_actionsd(), dims.oP(),
361 dims.oQ(), dims.oD(), this->m_handlers->rr(), this->m_handlers->rp(),
362 this->m_handlers->rq());
363 if (dims.nP > 0)
364 this->m_apply_metric_p(detail::construct_vectorP({row}, solutions, dims.oP(), dims.nP),
365 this->m_xspace->cparamsp(), wrap_arg(metric_action));
366 }
