256 void construct_residual(const std::vector<int>& roots, const CVecRef<R>& params, const VecRef<R>& actions,
257 const detail::MetricSolutions<R, Q>& metric_solutions) override {
258 if (roots.empty())
259 return;
260 auto prof = this->profiler()->push("itsolv::construct_residual");
261 assert(params.size() >= roots.size());
262 const auto& eigvals = eigenvalues();
263
264 auto subspace_solver = std::dynamic_pointer_cast<subspace::SubspaceSolverLinEig<R, Q, P>>(this->m_subspace_solver);
265 assert(subspace_solver);
266 const auto& imag_eigval_components = subspace_solver->imag_eigval_components();
267
268 // With a metric M, the residual contains M x rather than x for each solution x. M x is given in metric_solutions
269 // where the metric is applied on the fly, and otherwise assembled from the stored actions of the metric on the Q
270 // and D spaces and the action of the metric on the P space. The coefficients are collected in these matrices.
271 const auto& xspace = *this->m_xspace;
272 const bool metric = xspace.metric_kind() != MetricKind::Identity;
273 const bool on_the_fly = metric && xspace.metric_storage() == MetricStorage::OnTheFly;
274 const bool stored = metric && !on_the_fly;
275 const auto& dims = xspace.dimensions();
276 const auto& solutions = this->m_subspace_solver->solutions();
277 using coefficient_matrix = std::decay_t<decltype(solutions)>;
278 // the coefficients of the Q and D vectors are in one matrix, with the Q vectors first
279 coefficient_matrix mp_coeff({stored ? dims.nP : 0, roots.size()}),
280 mqd_coeff({stored ? dims.nQ + dims.nD : 0, roots.size()}),
281 mx_coeff({on_the_fly ? roots.size() : 0, roots.size()});
282 // adds weight times the action of the metric on the solution params[k], whose coefficients are in the given row of
283 // coefficients, to the residual actions[i]
284 auto add_metric_solution = [&](const scalar_type weight, const coefficient_matrix& coefficients, const int row,
285 const size_t k, const size_t i) {
286 if (on_the_fly) {
287 mx_coeff(k, i) += weight;
288 } else {
289 for (size_t j = 0; j < dims.nP; ++j)
290 mp_coeff(j, i) += weight * coefficients(row, dims.oP() + j);
291 for (size_t j = 0; j < dims.nQ; ++j)
292 mqd_coeff(j, i) += weight * coefficients(row, dims.oQ() + j);
293 for (size_t j = 0; j < dims.nD; ++j)
294 mqd_coeff(dims.nQ + j, i) += weight * coefficients(row, dims.oD() + j);
295 }
296 };
297
298 for (size_t i = 0; i < roots.size(); ++i) {
299 if (metric)
300 add_metric_solution(-eigvals.at(roots[i]), solutions, roots[i], i, i);
301 else
302 this->m_handlers->rr().axpy(-eigvals.at(roots[i]), params.at(i), actions.at(i));
303
304 auto it = std::ranges::find(imag_eigval_components, roots[i], [](const auto& pair) { return pair.first; });
305
306 if (it != imag_eigval_components.end()) {
307 // Ref.: https://doi.org/10.1063/1.2755681 (Appendix)
308 // The idea is the following: We make use of the fact that the real and imaginary parts of the
309 // eigenvectors belonging to the complex root pair are spanned by the same basis. This allows
310 // us to split the associated residual vectors into a part corresponding to the real and a part
311 // corresponding to the imaginary part as well. The actual (complex-valued) residual can then
312 // always be reconstructed from this and therefore we don't lose any information. The formulas are
313 // |(R_real)_i> = sum_j [ (|A_j> - (e_real)_i |b_j>) (c_real)_{ji} + (e_imag) |b_j> (c_imag)_{ji} ]
314 // |(R_imag)_i> = sum_j [ (|A_j> - (e_real)_i |b_j>) (c_imag)_{ji} - (e_imag) |b_j> (c_real)_{ji} ]
315 // where |A_j> is the j-th action and |b_j> the j-th trial vector. e are the complex conjugate
316 // eigenvalues and c the associated eigenvectors.
317 // Note: At this point, params and actions already contain the vectors transformed into the
318 // eigenbasis (aka.: multiplied with c). Due to the way we process these complex-valued vectors,
319 // the first vector in a pair is the one that has been transformed with the real and the second
320 // with the imaginary part of c.
321 // The above code has already dealt with the bulk of the necessary expression and we only need
322 // to add the part with the imaginary components of the eigenvalue pair, for which the vector of
323 // the partner root is used.
324 // Note: The minus sign in above expression is implicitly taken into account as the
325 // imaginary parts of the complex conjugate eigenvalue pair has flipped signs
326 const auto partner = subspace_solver->complex_partner(roots[i]).value();
327 const auto root_it = std::ranges::find(roots, int(partner));
328 if (root_it != roots.end() && (!metric || on_the_fly)) {
329 const auto k = std::ranges::distance(roots.begin(), root_it);
330 if (metric)
331 add_metric_solution(it->second, solutions, int(partner), k, i);
332 else
333 this->m_handlers->rr().axpy(it->second, params.at(k), actions.at(i));
334 } else if (stored) {
335 // The partner's coefficients are among the retained solutions also if it is not among roots
336 if (partner >= subspace_solver->retained_solutions().rows())
337 throw std::logic_error("The complex conjugate partner of a root is not among the retained solutions");
338 add_metric_solution(it->second, subspace_solver->retained_solutions(), int(partner), 0, i);
339 } else {
340 // The partner is not among roots, or was truncated away, so its vector is assembled from the subspace
341 if (partner >= subspace_solver->retained_solutions().rows())
342 throw std::logic_error("The complex conjugate partner of a root is not among the retained solutions");
343 auto partner_param = this->m_handlers->rr().copy(actions.at(i));
344 construct_subspace_vector(subspace_solver->retained_solutions(), int(partner), partner_param);
345 if (on_the_fly)
346 apply_metric(partner_param);
347 this->m_handlers->rr().axpy(it->second, partner_param, actions.at(i));
348 }
349 }
350 }
351 const auto residuals = wrap(actions.begin(), actions.begin() + roots.size());
352 if (stored) {
353 if (dims.nP > 0) {
354 using VectorP = typename SolverTemplate::VectorP;
355 auto pvectors = std::vector<VectorP>(roots.size(), VectorP(dims.nP));
356 for (size_t i = 0; i < roots.size(); ++i)
357 for (size_t j = 0; j < dims.nP; ++j)
358 pvectors[i][j] = mp_coeff(j, i);
359 this->m_apply_metric_p(pvectors, xspace.cparamsp(), residuals);
360 }
361 // one pass over the residuals for the actions of the metric on the Q and D vectors
362 auto qd_metric_actions = xspace.cmetric_actionsq();
363 const auto d_metric_actions = xspace.cmetric_actionsd();
364 qd_metric_actions.insert(qd_metric_actions.end(), d_metric_actions.begin(), d_metric_actions.end());
365 this->m_handlers->rq().gemm_outer(mqd_coeff, qd_metric_actions, residuals);
366 } else if (on_the_fly && !metric_solutions.r.empty()) {
367 this->m_handlers->rr().gemm_outer(mx_coeff, metric_solutions.r, residuals);
368 } else if (on_the_fly) {
369 this->m_handlers->rq().gemm_outer(mx_coeff, metric_solutions.q, residuals);
370 }
371 }

379 void combine_errors() override {
380 auto subspace_solver = std::dynamic_pointer_cast<subspace::SubspaceSolverLinEig<R, Q, P>>(this->m_subspace_solver);
381 assert(subspace_solver);
382 const auto& errors = subspace_solver->errors();
383 for (size_t root = 0; root < subspace_solver->size(); ++root) {
384 const auto partner = subspace_solver->complex_partner(root);
385 if (!partner || *partner <= root || *partner >= subspace_solver->size())
386 continue;
387 const auto error = std::sqrt(errors[root] * errors[root] + errors[*partner] * errors[*partner]);
388 subspace_solver->set_error(int(root), error);
389 subspace_solver->set_error(int(*partner), error);
390 }
391 }

400 void compute_errors(const std::vector<int>& roots, const CVecRef<R>& params, const CVecRef<R>& residuals,
401 const std::vector<value_type_abs>& solution_norms, std::vector<value_type_abs>& errors) override {
402 SolverTemplate::compute_errors(roots, params, residuals, solution_norms, errors);
403 auto subspace_solver = std::dynamic_pointer_cast<subspace::SubspaceSolverLinEig<R, Q, P>>(this->m_subspace_solver);
404 assert(subspace_solver);
405 const auto nroots = subspace_solver->size();
406 if (nroots == 0)
407 return;
408 const auto partner = subspace_solver->complex_partner(nroots - 1);
409 const auto root_it = std::ranges::find(roots, int(nroots - 1));
410 if (!partner || *partner < nroots || root_it == roots.end())
411 return;
412 const auto i = std::ranges::distance(roots.begin(), root_it);
413 const auto& imag_eigval_components = subspace_solver->imag_eigval_components();
414 const auto it = std::ranges::find(imag_eigval_components, nroots - 1, [](const auto& pair) { return pair.first; });
415 if (it == imag_eigval_components.end())
416 throw std::logic_error("The imaginary part of the eigenvalue of a complex pair member is missing");
417 const auto& solutions = subspace_solver->retained_solutions();
418 auto partner_param = this->m_handlers->rr().copy(params.at(i));
419 auto partner_residual = this->m_handlers->rr().copy(params.at(i));
420 construct_subspace_vector(solutions, int(*partner), partner_param);
421 construct_subspace_action(solutions, int(*partner), partner_residual);
422 if (this->m_xspace->metric_kind() != MetricKind::Identity) {
423 // With a metric, the residual contains the actions of the metric on the partner and on the last root
424 auto metric_param = this->m_handlers->rr().copy(params.at(i));
425 if (this->m_xspace->stores_metric_actions()) {
426 construct_subspace_metric_action(solutions, int(*partner), partner_param);
427 construct_subspace_metric_action(solutions, int(nroots - 1), metric_param);
428 } else {
429 apply_metric(partner_param);
430 apply_metric(metric_param);
431 }
432 this->m_handlers->rr().axpy(-eigenvalues().at(nroots - 1), partner_param, partner_residual);
433 this->m_handlers->rr().axpy(-it->second, metric_param, partner_residual);
434 } else {
435 // The imaginary part of the partner's eigenvalue is the negative of the one of the last root
436 this->m_handlers->rr().axpy(-eigenvalues().at(nroots - 1), partner_param, partner_residual);
437 this->m_handlers->rr().axpy(-it->second, params.at(i), partner_residual);
438 }
439 auto partner_error = std::sqrt(std::abs(this->m_handlers->rr().dot(partner_residual, partner_residual)));
440 if (!solution_norms.empty())
441 // as in SolverTemplate::compute_errors(), the error of the partner of unit length
442 partner_error /=
443 subspace::util::solution_norms(solutions, this->m_xspace->data.at(subspace::EqnData::S))[*partner];
444 errors.at(i) = std::sqrt(errors.at(i) * errors.at(i) + partner_error * partner_error);
445 }

483 this->m_xspace->cparamsp(), wrap_arg(metric_action));
