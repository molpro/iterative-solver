115 void finalize() override {
116 if constexpr (is_complex<typename R::value_type>::value) {
117 return;
118 } else {
119 auto subspace_solver =
120 std::dynamic_pointer_cast<subspace::SubspaceSolverLinEig<R, Q, P>>(this->m_subspace_solver);
121 assert(subspace_solver);
122 auto imag_eigval_components = subspace_solver->imag_eigval_components();
123 if (imag_eigval_components.empty()) {
124 return;
125 }
126
127 // Convert to 1-based indexing for printout
128 for (auto& pair : imag_eigval_components) {
129 pair.first += 1;
130 }
131
132 this->m_logger->template warn<log::ComplexRootsDavidson<typename R::value_type>>(
133 "The following roots are complex-valued. Associated eigenvectors are the real and imaginary part "
134 "of the pairs and the imaginary parts of the eigenvalues are ",
135 imag_eigval_components);
136
137 const auto nroots = subspace_solver->size();
138 if (const auto partner = subspace_solver->complex_partner(nroots - 1); partner && *partner >= nroots)
139 this->m_logger->warn(std::format("The complex conjugate partner of root {} is not among the requested roots, so "
140 "only the real part of its eigenvector is available",
141 nroots));
142 }

242 void construct_residual(const std::vector<int>& roots, const CVecRef<R>& params, const VecRef<R>& actions) override {
243 auto prof = this->profiler()->push("itsolv::construct_residual");
244 assert(params.size() >= roots.size());
245 const auto& eigvals = eigenvalues();
246
247 auto subspace_solver = std::dynamic_pointer_cast<subspace::SubspaceSolverLinEig<R, Q, P>>(this->m_subspace_solver);
248 assert(subspace_solver);
249 const auto& imag_eigval_components = subspace_solver->imag_eigval_components();
250
251 for (size_t i = 0; i < roots.size(); ++i) {
252 this->m_handlers->rr().axpy(-eigvals.at(roots[i]), params.at(i), actions.at(i));
253
254 auto it = std::ranges::find(imag_eigval_components, roots[i], [](const auto& pair) { return pair.first; });
255
256 if (it != imag_eigval_components.end()) {
257 // Ref.: https://doi.org/10.1063/1.2755681 (Appendix)
258 // The idea is the following: We make use of the fact that the real and imaginary parts of the
259 // eigenvectors belonging to the complex root pair are spanned by the same basis. This allows
260 // us to split the associated residual vectors into a part corresponding to the real and a part
261 // corresponding to the imaginary part as well. The actual (complex-valued) residual can then
262 // always be reconstructed from this and therefore we don't lose any information. The formulas are
263 // |(R_real)_i> = sum_j [ (|A_j> - (e_real)_i |b_j>) (c_real)_{ji} + (e_imag) |b_j> (c_imag)_{ji} ]
264 // |(R_imag)_i> = sum_j [ (|A_j> - (e_real)_i |b_j>) (c_imag)_{ji} - (e_imag) |b_j> (c_real)_{ji} ]
265 // where |A_j> is the j-th action and |b_j> the j-th trial vector. e are the complex conjugate
266 // eigenvalues and c the associated eigenvectors.
267 // Note: At this point, params and actions already contain the vectors transformed into the
268 // eigenbasis (aka.: multiplied with c). Due to the way we process these complex-valued vectors,
269 // the first vector in a pair is the one that has been transformed with the real and the second
270 // with the imaginary part of c.
271 // The above code has already dealt with the bulk of the necessary expression and we only need
272 // to add the part with the imaginary components of the eigenvalue pair, for which the vector of
273 // the partner root is used.
274 // Note: The minus sign in above expression is implicitly taken into account as the
275 // imaginary parts of the complex conjugate eigenvalue pair has flipped signs
276 const auto partner = subspace_solver->complex_partner(roots[i]).value();
277 const auto root_it = std::ranges::find(roots, int(partner));
278 if (root_it != roots.end()) {
279 this->m_handlers->rr().axpy(it->second, params.at(std::ranges::distance(roots.begin(), root_it)),
280 actions.at(i));
281 } else {
282 // The partner is not among roots, or was truncated away, so its vector is assembled from the subspace
283 if (partner >= subspace_solver->retained_solutions().rows())
284 throw std::logic_error("The complex conjugate partner of a root is not among the retained solutions");
285 auto partner_param = this->m_handlers->rr().copy(actions.at(i));
286 construct_subspace_vector(subspace_solver->retained_solutions(), int(partner), partner_param);
287 this->m_handlers->rr().axpy(it->second, partner_param, actions.at(i));
288 }
289 }
290 }
291 }

299 void combine_errors() override {
300 auto subspace_solver = std::dynamic_pointer_cast<subspace::SubspaceSolverLinEig<R, Q, P>>(this->m_subspace_solver);
301 assert(subspace_solver);
302 const auto& errors = subspace_solver->errors();
303 for (size_t root = 0; root < subspace_solver->size(); ++root) {
304 const auto partner = subspace_solver->complex_partner(root);
305 if (!partner || *partner <= root || *partner >= subspace_solver->size())
306 continue;
307 const auto error = std::sqrt(errors[root] * errors[root] + errors[*partner] * errors[*partner]);
308 subspace_solver->set_error(int(root), error);
309 subspace_solver->set_error(int(*partner), error);
310 }
311 }

320 void compute_errors(const std::vector<int>& roots, const CVecRef<R>& params, const CVecRef<R>& residuals,
321 std::vector<value_type_abs>& errors) override {
322 SolverTemplate::compute_errors(roots, params, residuals, errors);
323 auto subspace_solver = std::dynamic_pointer_cast<subspace::SubspaceSolverLinEig<R, Q, P>>(this->m_subspace_solver);
324 assert(subspace_solver);
325 const auto nroots = subspace_solver->size();
326 if (nroots == 0)
327 return;
328 const auto partner = subspace_solver->complex_partner(nroots - 1);
329 const auto root_it = std::ranges::find(roots, int(nroots - 1));
330 if (!partner || *partner < nroots || root_it == roots.end())
331 return;
332 const auto i = std::ranges::distance(roots.begin(), root_it);
333 const auto& imag_eigval_components = subspace_solver->imag_eigval_components();
334 const auto it = std::ranges::find(imag_eigval_components, nroots - 1, [](const auto& pair) { return pair.first; });
335 if (it == imag_eigval_components.end())
336 throw std::logic_error("The imaginary part of the eigenvalue of a complex pair member is missing");
337 const auto& solutions = subspace_solver->retained_solutions();
338 auto partner_param = this->m_handlers->rr().copy(params.at(i));
339 auto partner_residual = this->m_handlers->rr().copy(params.at(i));
340 construct_subspace_vector(solutions, int(*partner), partner_param);
341 construct_subspace_action(solutions, int(*partner), partner_residual);
342 // The imaginary part of the partner's eigenvalue is the negative of the one of the last root
343 this->m_handlers->rr().axpy(-eigenvalues().at(nroots - 1), partner_param, partner_residual);
344 this->m_handlers->rr().axpy(-it->second, params.at(i), partner_residual);
345 errors.at(i) = std::sqrt(errors.at(i) * errors.at(i) +
346 std::abs(this->m_handlers->rr().dot(partner_residual, partner_residual)));
347 }
