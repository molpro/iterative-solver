49template <class R, class Q, class P>
50auto update_qspace_data(const CVecRef<R>& params, const CVecRef<R>& actions, const CVecRef<P>& pparams,
51 const CVecRef<Q>& qparams, const CVecRef<Q>& qactions, const CVecRef<Q>& dparams,
52 const CVecRef<Q>& dactions, const CVecRef<Q>& rhs, const Dimensions& dims,
53 ArrayHandlers<R, Q, P>& handlers, Logger& logger, bool hermitian = false,
54 bool action_dot_action = false, const CVecRef<R>& metric_actions = {}) {
55 auto nQnew = params.size();
56 const bool metric = !metric_actions.empty();
57 auto data = NewData<typename array::ArrayHandler<R, R>::value_type>(nQnew, dims.nX(), rhs.size(), metric);
58 auto& qq = data.qq;
59 auto& qx = data.qx;
60 auto& xq = data.xq;
61 qq[EqnData::S] = util::overlap(params, handlers.rr());
62 // rows nQnew and beyond of the overlaps with the subspace vectors are those of the actions of the metric
63 auto left = params;
64 left.insert(left.end(), metric_actions.begin(), metric_actions.end());
65 auto copy_overlaps = [&](const auto& overlaps, const size_t offset, const size_t size) {
66 qx[EqnData::S].slice({0, offset}, {nQnew, offset + size}) = overlaps.slice({0, 0}, {nQnew, size});
67 if (metric)
68 qx[EqnData::M].slice({0, offset}, {nQnew, offset + size}) = overlaps.slice({nQnew, 0}, {2 * nQnew, size});
69 };
70 copy_overlaps(util::overlap(left, pparams, handlers.rp()), dims.oP(), dims.nP);
71 copy_overlaps(util::overlap(left, qparams, handlers.rq()), dims.oQ(), dims.nQ);
72 copy_overlaps(util::overlap(left, dparams, handlers.rq()), dims.oD(), dims.nD);
73 qq[EqnData::H] =
74 action_dot_action ? util::overlap(actions, handlers.rr()) : util::overlap(params, actions, handlers.rr());
75 // left-hand vectors of the H matrix elements with existing actions
76 const auto& h_left = action_dot_action ? actions : params;
77 qx[EqnData::H].slice({0, dims.oQ()}, {nQnew, dims.oQ() + dims.nQ}) = util::overlap(h_left, qactions, handlers.rq());
78 qx[EqnData::H].slice({0, dims.oD()}, {nQnew, dims.oD() + dims.nD}) = util::overlap(h_left, dactions, handlers.rq());
79 if (hermitian) {
80 xq[EqnData::H].slice({dims.oP(), 0}, {dims.oP() + dims.nP, nQnew}) = util::overlap(pparams, actions, handlers.rp());
81 // xq[EqnData::H].slice({dims.oQ(), 0}, {dims.oQ() + dims.nQ, nQnew}) = util::overlap(action_dot_action ? qactions :
82 // qparams, actions, handlers.qr());
83 conjugate_transpose_copy(xq[EqnData::H].slice({dims.oQ(), 0}, {dims.oQ() + dims.nQ, nQnew}),
84 qx[EqnData::H].slice({0, dims.oQ()}, {nQnew, dims.oQ() + dims.nQ}));
85 conjugate_transpose_copy(xq[EqnData::H].slice({dims.oD(), 0}, {dims.oD() + dims.nD, nQnew}),
86 qx[EqnData::H].slice({0, dims.oD()}, {nQnew, dims.oD() + dims.nD}));
87 conjugate_transpose_copy(qx[EqnData::H].slice({0, dims.oP()}, {nQnew, dims.oP() + dims.nP}),
88 xq[EqnData::H].slice({dims.oP(), 0}, {dims.oP() + dims.nP, nQnew}));
89 } else {
90 xq[EqnData::H].slice({dims.oQ(), 0}, {dims.oQ() + dims.nQ, nQnew}) = util::overlap(qparams, actions, handlers.rq());
91 xq[EqnData::H].slice({dims.oD(), 0}, {dims.oD() + dims.nD, nQnew}) = util::overlap(dparams, actions, handlers.rq());
92 }
93 if (metric) {
94 qq[EqnData::M] = util::overlap(params, metric_actions, handlers.rr());
95 conjugate_transpose_copy(xq[EqnData::M].slice(), qx[EqnData::M].slice());
96 }
97 qq[EqnData::rhs] = util::overlap(params, rhs, handlers.rq());
98 conjugate_transpose_copy(xq[EqnData::S].slice({dims.oP(), 0}, {dims.oP() + dims.nP, nQnew}),
99 qx[EqnData::S].slice({0, dims.oP()}, {nQnew, dims.oP() + dims.nP}));
100 conjugate_transpose_copy(xq[EqnData::S].slice({dims.oQ(), 0}, {dims.oQ() + dims.nQ, nQnew}),
101 qx[EqnData::S].slice({0, dims.oQ()}, {nQnew, dims.oQ() + dims.nQ}));
102 conjugate_transpose_copy(xq[EqnData::S].slice({dims.oD(), 0}, {dims.oD() + dims.nD, nQnew}),
103 qx[EqnData::S].slice({0, dims.oD()}, {nQnew, dims.oD() + dims.nD}));
104 logger.data_dump("xspace::update_qspace_data() nQnew = ", nQnew);
105 logger.data_dump("Sqq = ", qq[EqnData::S]);
106 logger.data_dump("Hqq = ", qq[EqnData::H]);
107 logger.data_dump("Sqx = ", qx[EqnData::S]);
108 logger.data_dump("Hqx = ", qx[EqnData::H]);
109 logger.data_dump("Sxq = ", xq[EqnData::S]);
110 logger.data_dump("Hxq = ", xq[EqnData::H]);
111 if (metric) {
112 logger.data_dump("Mqq = ", qq[EqnData::M]);
113 logger.data_dump("Mqx = ", qx[EqnData::M]);
114 }
115 logger.data_dump("rhs_q = ", qq[EqnData::rhs]);
116 return data;
117}

120template <class Q, class P>
121auto update_dspace_overlap_data(const CVecRef<P>& pparams, const CVecRef<Q>& qparams, const CVecRef<Q>& dparams,
122 const CVecRef<Q>& rhs, array::ArrayHandler<Q, P>& handler_qp,
123 array::ArrayHandler<Q, Q>& handler_qq, Logger& logger) {
124 const auto nP = pparams.size();
125 const auto nQ = qparams.size();
126 const auto nX = nP + nQ;
127 auto nD = dparams.size();
128 const auto nRHS = rhs.size();
129 auto data = NewData<typename array::ArrayHandler<Q, Q>::value_type>(nD, nX, nRHS);
130 data.qq[EqnData::S].slice() = util::overlap(dparams, handler_qq);
131 data.qx[EqnData::S].slice({0, 0}, {nD, nP}) = util::overlap(dparams, pparams, handler_qp);
132 data.qx[EqnData::S].slice({0, nP}, {nD, nX}) = util::overlap(dparams, qparams, handler_qq);
133 data.qq[EqnData::rhs].slice() = util::overlap(dparams, rhs, handler_qq);
134 conjugate_transpose_copy(data.xq[EqnData::S].slice(), data.qx[EqnData::S].slice());
135 logger.data_dump("xspace::update_dspace_overlap_data() nD = ", nD);
136 logger.data_dump("Sdd = ", data.qq[EqnData::S]);
137 logger.data_dump("Sdx = ", data.qx[EqnData::S]);
138 logger.data_dump("rhs_d = ", data.qq[EqnData::rhs]);
139 return data;
140}

143template <class Q, class P>
144auto update_dspace_action_data(const CVecRef<P>& pparams, const CVecRef<Q>& qparams, const CVecRef<Q>& qactions,
145 const CVecRef<Q>& dparams, const CVecRef<Q>& dactions,
146 array::ArrayHandler<Q, P>& handler_qp, array::ArrayHandler<Q, Q>& handler_qq,
147 Logger& logger) {
148 const auto nP = pparams.size();
149 const auto nQ = qparams.size();
150 const auto nX = nP + nQ;
151 auto nD = dparams.size();
152 auto data = NewData<typename array::ArrayHandler<Q, Q>::value_type>(nD, nX, 0);
153 const auto e = EqnData::H;
154 data.qq[e].slice() = util::overlap(dparams, dactions, handler_qq);
155 data.xq[e].slice({0, 0}, {nP, nD}) = util::overlap(pparams, dactions, handler_qp);
156 data.xq[e].slice({nP, 0}, {nX, nD}) = util::overlap(qparams, dactions, handler_qq);
157 data.qx[e].slice({0, nP}, {nD, nX}) = util::overlap(dparams, qactions, handler_qq);
158 conjugate_transpose_copy(data.qx[e].slice({0, 0}, {nD, nP}), data.xq[e].slice({0, 0}, {nP, nD}));
159 logger.data_dump("xspace::update_dspace_action_data() nD = ", nD);
160 logger.data_dump("Hdd = ", data.qq[e]);
161 logger.data_dump("Hdx = ", data.qx[e]);
162 logger.data_dump("Hxd = ", data.xq[e]);
163 return data;
164}

175template <typename value_type>
176Matrix<value_type> metric_with_new_dspace(const Matrix<value_type>& metric, const Dimensions& dims,
177 const std::vector<std::size_t>& q_delete,
178 const Matrix<value_type>& dcoefficients) {
179 using matrix_type = Eigen::Matrix<value_type, Eigen::Dynamic, Eigen::Dynamic>;
180 // the kept P and Q vectors, and the deleted Q and old D vectors that the new D vectors are combinations of
181 auto kept = std::vector<size_t>{}, combined = std::vector<size_t>{};
182 for (size_t i = 0; i < dims.nP; ++i)
183 kept.push_back(dims.oP() + i);
184 for (size_t i = 0; i < dims.nQ; ++i)
185 if (std::find(q_delete.begin(), q_delete.end(), i) == q_delete.end())
186 kept.push_back(dims.oQ() + i);
187 for (auto iq : q_delete)
188 combined.push_back(dims.oQ() + iq);
189 for (size_t i = 0; i < dims.nD; ++i)
190 combined.push_back(dims.oD() + i);
191 const auto nK = kept.size(), nB = combined.size(), nDnew = dcoefficients.rows();
192 // only the blocks involving the new D vectors need multiplications, of cost nX * nB * nDnew
193 matrix_type m_kb(nK, nB), m_bb(nB, nB), c(nDnew, nB);
194 for (size_t k = 0; k < nK; ++k)
195 for (size_t j = 0; j < nB; ++j)
196 m_kb(k, j) = metric(kept[k], combined[j]);
197 for (size_t i = 0; i < nB; ++i)
198 for (size_t j = 0; j < nB; ++j)
199 m_bb(i, j) = metric(combined[i], combined[j]);
200 for (size_t i = 0; i < nDnew; ++i)
201 for (size_t j = 0; j < nB; ++j)
202 c(i, j) = dcoefficients(i, j);
203 // <k|M d_i> = sum_j <k|M b_j> c_ij, and <d_i|M d_l> = sum_jm conj(c_ij) <b_j|M b_m> c_lm
204 const matrix_type m_kd = m_kb * c.transpose();
205 const matrix_type m_dd = c.conjugate() * m_bb * c.transpose();
206 const auto nXnew = nK + nDnew;
207 auto m_new = Matrix<value_type>({nXnew, nXnew});
208 for (size_t k = 0; k < nK; ++k)
209 for (size_t l = 0; l < nK; ++l)
210 m_new(k, l) = metric(kept[k], kept[l]);
211 for (size_t k = 0; k < nK; ++k)
212 for (size_t i = 0; i < nDnew; ++i) {
213 m_new(k, nK + i) = m_kd(k, i);
214 m_new(nK + i, k) = conjugate(m_kd(k, i));
215 }
216 for (size_t i = 0; i < nDnew; ++i)
217 for (size_t l = 0; l < nDnew; ++l)
218 m_new(nK + i, nK + l) = m_dd(i, l);
219 return m_new;
220}

222template <typename T>
223void copy_dspace_eqn_data(const NewData<T>& new_data, SubspaceData<T>& data, const subspace::EqnData e,
224 const Dimensions& dims) {
225 const auto& dd = new_data.qq.at(e);
226 const auto& dx = new_data.qx.at(e);
227 const auto& xd = new_data.xq.at(e);
228 data[e].slice({dims.oD(), dims.oD()}, {dims.oD() + dims.nD, dims.oD() + dims.nD}) = dd;
229 data[e].slice({dims.oD(), dims.oP()}, {dims.oD() + dims.nD, dims.oP() + dims.nP}) =
230 dx.slice({0, 0}, {dims.nD, dims.nP});
231 data[e].slice({dims.oD(), dims.oQ()}, {dims.oD() + dims.nD, dims.oQ() + dims.nQ}) =
232 dx.slice({0, dims.nP}, {dims.nD, dims.nP + dims.nQ});
233 data[e].slice({dims.oP(), dims.oD()}, {dims.oP() + dims.nP, dims.oD() + dims.nD}) =
234 xd.slice({0, 0}, {dims.nP, dims.nD});
235 data[e].slice({dims.oQ(), dims.oD()}, {dims.oQ() + dims.nQ, dims.oD() + dims.nD}) =
236 xd.slice({dims.nP, 0}, {dims.nP + dims.nQ, dims.nD});
237}

241class XSpace : public IXSpace<R, Q, P> {
242public:

248 explicit XSpace(const std::shared_ptr<ArrayHandlers<R, Q, P>>& handlers, const std::shared_ptr<Logger>& logger)
249 : pspace(), qspace(handlers, logger), dspace(logger), m_handlers(handlers), m_logger(logger) {
250 data = null_data<value_type, EqnData::H, EqnData::S, EqnData::rhs>();

253 void set_metric(MetricKind kind, MetricStorage storage = MetricStorage::Stored) override {
254 if (kind == m_metric_kind && storage == m_metric_storage)
255 return;
256 if (m_dim.nX() != 0)
257 throw std::logic_error("XSpace::set_metric: the metric cannot be changed once the subspace is not empty");
258 m_metric_kind = kind;
259 m_metric_storage = storage;
260 if (metric_kind() != MetricKind::Identity)
261 data[EqnData::M] = {};
262 else
263 data.erase(EqnData::M);
264 }

266 MetricKind metric_kind() const override { return m_metric_kind; }
267 MetricStorage metric_storage() const override { return m_metric_storage; }

270 void update_qspace(const CVecRef<R>& params, const CVecRef<R>& actions,
271 const CVecRef<R>& metric_actions = {}) override {
272 m_logger->trace("QSpace::update_qspace");
273 if (metric_actions.size() != (metric_kind() != MetricKind::Identity ? params.size() : 0))
274 throw std::invalid_argument(
275 "XSpace::update_qspace: actions of the metric must be given for every parameter exactly when there is a "
276 "metric");
277 auto new_data = xspace::update_qspace_data(params, actions, cparamsp(), cparamsq(), cactionsq(), cparamsd(),
278 cactionsd(), cwrap(m_rhs), m_dim, *m_handlers, *m_logger, m_hermitian,
279 m_action_dot_action, metric_actions);
280 qspace.update(params, actions, new_data.qq, new_data.qx, new_data.xq, m_dim, data,
281 stores_metric_actions() ? metric_actions : CVecRef<R>{});
282 update_dimensions();
283 }

286 void update_dspace(VecRef<Q>& params, VecRef<Q>& actions, const VecRef<Q>& metric_actions = {},
287 const std::vector<size_t>& q_delete = {}, const Matrix<value_type>& coefficients = {}) override {
288 if (metric_actions.size() != (stores_metric_actions() ? params.size() : 0))
289 throw std::invalid_argument(
290 "XSpace::update_dspace: actions of the metric must be given exactly when they are stored");
291 if (metric_kind() != MetricKind::Identity && !params.empty() &&
292 (coefficients.rows() != params.size() || coefficients.cols() != q_delete.size() + m_dim.nD))
293 throw std::invalid_argument("XSpace::update_dspace: the coefficients of the new D vectors are needed");
294 // the metric matrix of the new subspace is transformed from the current one, before the subspace changes
295 const auto metric_matrix = metric_kind() != MetricKind::Identity && !params.empty()
296 ? xspace::metric_with_new_dspace(data.at(EqnData::M), m_dim, q_delete, coefficients)
297 : Matrix<value_type>{};
298 auto q_delete_descending = q_delete;
299 std::sort(q_delete_descending.begin(), q_delete_descending.end(), std::greater<>());
300 for (auto iq : q_delete_descending)
301 eraseq(iq);
302 dspace.update(params, actions, metric_actions);
303 update_dimensions();
304 for (auto e : square_blocks(data))
305 data[e].resize({m_dim.nX(), m_dim.nX()});
306 auto new_data = xspace::update_dspace_overlap_data(cparamsp(), cparamsq(), cparamsd(), cwrap(m_rhs),
307 m_handlers->qp(), m_handlers->qq(), *m_logger);
308 xspace::copy_dspace_eqn_data(new_data, data, EqnData::S, m_dim);
309 auto new_data_action = xspace::update_dspace_action_data(
310 cparamsp(), cparamsq(), cactionsq(), cparamsd(), cactionsd(), m_handlers->qp(), m_handlers->qq(), *m_logger);
311 xspace::copy_dspace_eqn_data(new_data_action, data, EqnData::H, m_dim);
312 if (metric_kind() != MetricKind::Identity && m_dim.nD > 0)
313 data[EqnData::M] = metric_matrix;
314 data[EqnData::rhs].resize({m_dim.nX(), m_dim.nRHS});
315 data[EqnData::rhs].slice({m_dim.oD(), 0}, {m_dim.oD() + m_dim.nD, m_dim.nRHS}) = new_data.qq[EqnData::rhs].slice();
316 }

320 void update_pspace(const CVecRef<P>& params, const array::Span<value_type>& pp_action_matrix,
321 const array::Span<value_type>& pp_metric_matrix = {}) override {
322 assert(m_dim.nX() == 0);

324 throw std::runtime_error("P space can only be used with hermitian kernels");
325 const size_t nPnew = params.size();
326 if (pp_metric_matrix.size() != (metric_kind() != MetricKind::Identity ? nPnew * nPnew : 0))
327 throw std::invalid_argument("XSpace::update_pspace: the metric matrix in the P space must be given exactly when "
328 "there is a metric");
329 pspace.update(params, m_handlers->pp());
330 update_dimensions();
331 const size_t nP = m_dim.nP;
332 update_rhs_with_pspace();
333 data[EqnData::S].resize({nP, nP});
334 data[EqnData::H].resize({nP, nP});
335 data[EqnData::S].slice() = util::overlap(params, m_handlers->pp());
336 for (size_t i = 0, ij = 0; i < nP; ++i)
337 for (size_t j = 0; j < nP; ++j, ++ij)
338 data[EqnData::H](i, j) = pp_action_matrix[ij];
339 if (metric_kind() != MetricKind::Identity) {
340 data[EqnData::M].resize({nP, nP});
341 for (size_t i = 0, ij = 0; i < nP; ++i)
342 for (size_t j = 0; j < nP; ++j, ++ij)
343 data[EqnData::M](i, j) = pp_metric_matrix[ij];
344 }

348 void add_rhs_equations(const CVecRef<R>& rhs) {
349 m_rhs.reserve(m_rhs.size() + rhs.size());
350 for (const auto& r : rhs)
351 m_rhs.emplace_back(this->m_handlers->qr().copy(r));
352 for (const auto& r : rhs) {
353 auto d = std::abs(this->m_handlers->rr().dot(r, r));
354 if (d == 0)
355 throw std::runtime_error("RHS vector cannot be zero");
356 m_rhs_norm.emplace_back(std::sqrt(d));
357 }
358 update_dimensions();
359 update_rhs_with_pspace();
360 }
