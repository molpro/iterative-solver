40// Divides each action[k] elementwise by (diagonals - shift[k] * metric_diagonals), where metric_diagonals == nullptr
41// stands for a unit metric; if regularise, a small constant is added to the denominator
42template <typename T, typename S,
43 typename = std::enable_if_t<std::is_base_of<molpro::linalg::array::DistrArray, T>::value>>
44void precondition_default(const VecRef<T>& action, const std::vector<S>& shift, const T& diagonals,
45 const T* metric_diagonals, const bool regularise) {
46 // regularisation of the denominator, calibrated for double precision and rescaled to the working precision
47 const auto regulariser = regularise ? precision_scaled<S>(1e-15) : decltype(precision_scaled<S>(1e-15))(0);
48 auto diagonals_local_buffer = diagonals.local_buffer();
49 auto metric_local_buffer = metric_diagonals ? metric_diagonals->local_buffer() : nullptr;
50 for (size_t k = 0; k < action.size(); k++) {
51 auto action_local_buffer = action[k].get().local_buffer();
52 auto& distribution = action[k].get().distribution();
53 auto range = distribution.range(molpro::mpi::rank_global());
54 for (auto i = range.first; i < range.second; i++) {
55 const auto j = i - range.first;
56 const auto scaled_shift = metric_local_buffer ? shift[k] * (*metric_local_buffer)[j] : shift[k];
57 (*action_local_buffer)[j] /= ((*diagonals_local_buffer)[j] - scaled_shift + regulariser);
58 }
59 }
60}

62template <class T, typename S>
63void precondition_default(const VecRef<T>& action, const std::vector<S>& shift, const T& diagonals,
64 const T* metric_diagonals, const bool regularise,
65 typename T::iterator* = nullptr // SFINAE
66) {
67 const auto regulariser = regularise ? precision_scaled<S>(1e-15) : decltype(precision_scaled<S>(1e-15))(0);
68 for (size_t k = 0; k < action.size(); k++) {
69 auto a = action[k].get().begin();
70 auto m = metric_diagonals ? metric_diagonals->begin() : decltype(diagonals.begin()){};
71 for (auto d = diagonals.begin(); d != diagonals.end(); ++d, ++a) {
72 const auto scaled_shift = metric_diagonals ? shift[k] * *m++ : shift[k];
73 *a /= (*d - scaled_shift + regulariser);
74 }
75 }
76}
