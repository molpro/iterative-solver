110 std::vector<value_type> values() const override {
111 if (get_augmented_hessian() > 0)
112 return this->m_subspace_solver->eigenvalues();
113 const auto& solutions = this->m_subspace_solver->solutions(); // c(i, k): root i, subspace vector k
114 const auto& rhs = this->m_xspace->data.at(subspace::EqnData::rhs); // <v_k|b_i>
115 auto chi = std::vector<value_type>(std::min(solutions.rows(), rhs.cols()), value_type{0});
116 for (size_t i = 0; i < chi.size(); ++i)
117 for (size_t k = 0; k < std::min(solutions.cols(), rhs.rows()); ++k)
118 chi[i] += solutions(i, k) * molpro::linalg::conjugate(rhs(k, i));
119 return chi;
120 }

122 // FIXME The scale is fixed by the norm of RHS, but if RHS=0 there is no reference. We could use the norm of params
123 void construct_residual(const std::vector<int>& roots, const CVecRef<R>& params, const VecRef<R>& actions,
124 const detail::MetricSolutions<R, Q>& metric_solutions) override {
125 assert(params.size() >= roots.size());
126 const auto& norm = std::dynamic_pointer_cast<subspace::XSpace<R, Q, P>>(this->m_xspace)->rhs_norm();
127 // With the augmented Hessian the subspace solution satisfies (H - lambda) x = rhs, with lambda the subspace
128 // eigenvalue; without it lambda is zero
129 const auto& eigenvalues = this->m_subspace_solver->eigenvalues();
130 for (size_t i = 0; i < roots.size(); ++i) {
131 const auto ii = roots[i];
132 this->m_handlers->rq().axpy(-1, rhs().at(ii), actions.at(i));
133 if (size_t(ii) < eigenvalues.size() and eigenvalues[ii] != value_type(0))
134 this->m_handlers->rr().axpy(-eigenvalues[ii], params.at(i), actions.at(i));
135 if (norm.at(ii) != 0) {
136 auto scal = 1 / norm[ii];
137 this->m_handlers->rr().scal(scal, actions.at(i));
138 }
139 }
140 }
