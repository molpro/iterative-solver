234template <typename value_type>
235int eigensolver_hermitian_kernel(std::false_type /*use_lapack*/, std::span<value_type> a,
236 std::span<real_type_t<value_type>> w, size_t dimension) {
237 using matrix_type = Eigen::Matrix<value_type, Eigen::Dynamic, Eigen::Dynamic, Eigen::ColMajor>;
238 using real_vector_type = Eigen::Vector<real_type_t<value_type>, Eigen::Dynamic>;
239 Eigen::Map<matrix_type> A(a.data(), dimension, dimension);
240 // Eigen, like ?syev with uplo='L', references only the lower triangle and returns ascending eigenvalues
241 Eigen::SelfAdjointEigenSolver<matrix_type> solver(A, Eigen::ComputeEigenvectors);
242 if (solver.info() != Eigen::Success)
243 return 1;
244 Eigen::Map<real_vector_type>(w.data(), dimension) = solver.eigenvalues();
245 A = solver.eigenvectors();
246 return 0;
247}

267#endif

279template <typename value_type>
280int eigensolver_hermitian_lowest_kernel(std::false_type /*use_lapack*/, std::span<value_type> a,
281 std::span<value_type> z, std::span<real_type_t<value_type>> w,
282 size_t dimension, size_t nroots) {
283 std::vector<real_type_t<value_type>> all(dimension);
284 const int info = eigensolver_hermitian_kernel<value_type>(std::false_type{}, a, all, dimension);
285 std::copy_n(all.begin(), nroots, w.begin());
286 std::copy_n(a.begin(), dimension * nroots, z.begin());
287 return info;
288}

335template <typename value_type>
336int eigensolver_hermitian_lowest(std::span<const value_type> matrix, std::span<value_type> eigenvectors,
337 std::span<real_type_t<value_type>> eigenvalues, const size_t dimension,
338 const size_t nroots) {
339 if (matrix.size() != dimension * dimension || eigenvectors.size() != dimension * nroots ||
340 eigenvalues.size() != nroots || nroots > dimension) {
341 throw std::runtime_error("Sizes of matrix/eigenvectors/eigenvalues do not match dimension and nroots!");
342 }
343 if (nroots == 0)
344 return 0;
345 // the decomposition overwrites its input
346 std::vector<value_type> a(matrix.begin(), matrix.end());
347 return detail::eigensolver_hermitian_lowest_kernel<value_type>(has_lapack_kernel<value_type>{}, a, eigenvectors,
348 eigenvalues, dimension, nroots);
349}

353template <typename value_type>
354real_type_t<value_type> one_norm(std::span<const value_type> a, size_t dimension) {
355 using std::abs;
356 real_type_t<value_type> norm = 0;
357 for (size_t j = 0; j < dimension; ++j) {
358 real_type_t<value_type> column = 0;
359 for (size_t i = 0; i < dimension; ++i)
360 column += abs(a[i + j * dimension]);
361 norm = std::max(norm, column);
362 }
363 return norm;
364}

378template <typename value_type>
379bool eigenvalues_certainly_above(std::span<const value_type> matrix, size_t dimension,
380 real_type_t<value_type> threshold, real_type_t<value_type> margin = 1e4) {
381#ifdef HAVE_LAPACKE
382 if constexpr (has_lapack_kernel_v<value_type>) {
383 using real_t = real_type_t<value_type>;
384 if (dimension == 0)
385 return false;
386 constexpr char store_lower_triangle = 'L';
387 const auto n = lapack_int(dimension);
388 const real_t anorm = detail::one_norm<value_type>(matrix, dimension);
389 std::vector<value_type> factor(matrix.begin(), matrix.end());
390 if (lapack::potrf(LAPACK_COL_MAJOR, store_lower_triangle, n, factor.data(), n) != 0)
391 return false;
392 real_t rcond = 0;
393 if (lapack::pocon(LAPACK_COL_MAJOR, store_lower_triangle, n, factor.data(), n, anorm, &rcond) != 0)
394 return false;
395 return rcond * anorm > margin * threshold;
396 }
397#endif
398 return false;
399}

415template <typename value_type>
416bool generalized_eigensolver_hermitian_lowest(std::span<const value_type> matrix, std::span<const value_type> metric,
417 std::span<value_type> eigenvectors,
418 std::span<real_type_t<value_type>> eigenvalues, size_t dimension,
419 size_t nroots, real_type_t<value_type> min_rcond) {
420#ifdef HAVE_LAPACKE
421 if constexpr (has_lapack_kernel_v<value_type>) {
422 using real_t = real_type_t<value_type>;
423 if (nroots == 0 || nroots > dimension)
424 return false;
425 constexpr char store_lower_triangle = 'L';
426 const auto n = lapack_int(dimension);
427 const real_t anorm = detail::one_norm<value_type>(metric, dimension);
428 // the reduction overwrites both matrices, and leaves the Cholesky factor of S in place of S
429 std::vector<value_type> a(matrix.begin(), matrix.end()), b(metric.begin(), metric.end());
430 std::vector<lapack_int> ifail(dimension);
431 // ?sygvx/?hegvx need room for all dimension eigenvalues, though only the first nroots are returned
432 std::vector<real_t> all(dimension);
433 lapack_int found = 0;
434 if (lapack::hegvx_lowest(LAPACK_COL_MAJOR, store_lower_triangle, n, a.data(), n, b.data(), n, lapack_int(nroots),
435 &found, all.data(), eigenvectors.data(), n, ifail.data()) != 0 ||
436 size_t(found) != nroots)
437 return false;
438 std::copy_n(all.begin(), nroots, eigenvalues.begin());
439 real_t rcond = 0;
440 if (lapack::pocon(LAPACK_COL_MAJOR, store_lower_triangle, n, b.data(), n, anorm, &rcond) != 0)
441 return false;
442 return rcond >= min_rcond;
443 }
444#endif
445 return false;
446}

463template <typename value_type>
464int eigensolver_hermitian(std::span<const value_type> matrix, std::span<value_type> eigenvectors,
465 std::span<real_type_t<value_type>> eigenvalues, const size_t dimension) {
466 // validate input
467 if (eigenvectors.size() != matrix.size()) {
468 throw std::runtime_error("Matrix of eigenvectors and input matrix are not the same size! (" +
469 std::to_string(eigenvectors.size()) + " vs. " + std::to_string(matrix.size()) + ")");
470 }
471
472 if (eigenvectors.size() != dimension * dimension || eigenvalues.size() != dimension) {
473 throw std::runtime_error("Size of eigenvectors/eigenvalues do not match dimension!");
474 }
475
476 // copy input matrix, since the decomposition overwrites it in place
477 std::copy(matrix.begin(), matrix.end(), eigenvectors.begin());
478 if (dimension == 0)
479 return 0;
480
481 return detail::eigensolver_hermitian_kernel<value_type>(has_lapack_kernel<value_type>{}, eigenvectors, eigenvalues,
482 dimension);
483}

494template <typename value_type>
495std::list<SVD<value_type>> eigensolver_hermitian(size_t dimension, std::span<const value_type> matrix) {
496 using real_t = real_type_t<value_type>;
497 std::vector<value_type> eigvecs(dimension * dimension);
498 std::vector<real_t> eigvals(dimension);
499
500 const int success = eigensolver_hermitian<value_type>(matrix, eigvecs, eigvals, dimension);
501 if (success < 0) {
502 throw std::invalid_argument("Invalid argument of eigensolver_hermitian: " + std::to_string(-success));
503 }
504 if (success > 0) {
505 throw std::runtime_error("Hermitian eigensolver failed to converge. "
506 " elements of an intermediate tridiagonal form did not converge to zero.");
507 }
508
509 auto eigensystem = std::list<SVD<value_type>>{};
510
511 // populate eigensystem
512 for (int i = int(dimension) - 1; i >= 0;
513 i--) { // note: flipping this axis gives parity with results of eigen::jacobiSVD
514 auto temp_eigenproblem = SVD<value_type>{};
515 temp_eigenproblem.value = eigvals[i];
516 for (size_t j = 0; j < dimension; j++) {
517 temp_eigenproblem.v.emplace_back(eigvecs[j + (dimension * i)]);
518 }
519 eigensystem.emplace_back(temp_eigenproblem);
520 }
521
522 return eigensystem;
523}
