25 const size_t end, const value_type_abs norm_thresh) {
26 for (size_t i = start, j = start; i < end; ++i) {
27 if (norm[j] < norm_thresh) {

70template <class R, class Q, class P, typename value_type>
71void construct_solutions(const VecRef<R>& params, const std::vector<int>& roots,
72 const subspace::Matrix<value_type>& solutions, const CVecRef<P>& pparams,
73 const CVecRef<Q>& qparams, const CVecRef<Q>& dparams, size_t oP, size_t oQ, size_t oD,
74 array::ArrayHandler<R, R>& handler_rr, array::ArrayHandler<R, P>& handler_rp,
75 array::ArrayHandler<R, Q>& handler_rq) {
76 auto prof = molpro::Profiler::single();
77 prof->start("itsolv::util::construct_solutions");
78 assert(params.size() >= roots.size());
79 const auto nroots = roots.size();
80 for (size_t i = 0; i < nroots; ++i)
81 handler_rr.fill(0, params.at(i));
82 // coefficient matrices with the subspace vectors at the given indices in rows and roots in columns, as required by
83 // gemm_outer
84 auto coefficients = [&](const std::vector<size_t>& indices) {
85 auto coeff = subspace::Matrix<value_type>({indices.size(), nroots});
86 for (size_t i = 0; i < nroots; ++i)
87 for (size_t j = 0; j < indices.size(); ++j)
88 coeff(j, i) = solutions(roots[i], indices[j]);
89 return coeff;
90 };
91 auto append_indices = [](std::vector<size_t>& indices, size_t offset, size_t n) {
92 for (size_t j = 0; j < n; ++j)
93 indices.push_back(offset + j);
94 };
95 if (nroots > 0) {
96 const auto wparams = VecRef<R>(params.begin(), params.begin() + nroots);
97 auto p_indices = std::vector<size_t>{};
98 append_indices(p_indices, oP, pparams.size());
99 handler_rp.gemm_outer(coefficients(p_indices), pparams, wparams);
100 // Q and D in one call, so that each solution is passed over only once
101 auto qd_indices = std::vector<size_t>{};
102 append_indices(qd_indices, oQ, qparams.size());
103 append_indices(qd_indices, oD, dparams.size());
104 auto qdparams = qparams;
105 qdparams.insert(qdparams.end(), dparams.begin(), dparams.end());
106 handler_rq.gemm_outer(coefficients(qd_indices), qdparams, wparams);
107 }
108 prof->stop();
109}
