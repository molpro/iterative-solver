64template <class R>
65Matrix<typename array::ArrayHandler<R, R>::value_type> overlap(const CVecRef<R>& params,
66 array::ArrayHandler<R, R>& handler) {
67 auto m = Matrix<typename array::ArrayHandler<R, R>::value_type>({params.size(), params.size()});
68 for (size_t i = 0; i < m.rows(); ++i)
69 for (size_t j = 0; j <= i; ++j) {
70 m(i, j) = handler.dot(params[i], params[j]);
71 m(j, i) = conjugate(m(i, j)); // the matrix is hermitian, which for a real element type is symmetric
72 }
73 return m;
74}

81template <typename T>
82std::vector<real_type_t<T>> solution_norms(const Matrix<T>& solutions, const Matrix<T>& overlap) {
83 using row_major_type = Eigen::Matrix<T, Eigen::Dynamic, Eigen::Dynamic, Eigen::RowMajor>;
84 const auto nsol = Eigen::Index(solutions.rows()), dim = Eigen::Index(solutions.cols());
85 auto norms = std::vector<real_type_t<T>>(nsol);
86 if (nsol == 0 || dim == 0)
87 return norms;
88 const auto c = Eigen::Map<const row_major_type>(solutions.data().data(), nsol, dim);
89 const auto s = Eigen::Map<const row_major_type>(overlap.data().data(), dim, dim);
90 const row_major_type cs = c.conjugate() * s;
91 using std::abs;
92 using std::sqrt;
93 for (Eigen::Index i = 0; i < nsol; ++i)
94 norms[i] = sqrt(abs(cs.row(i).dot(c.row(i).conjugate())));
95 return norms;
96}

105template <typename T>
106Matrix<T> unit_length_solutions(const Matrix<T>& solutions, const Matrix<T>& overlap) {
107 auto scaled = solutions;
108 const auto norms = solution_norms(solutions, overlap);
109 for (size_t i = 0; i < scaled.rows(); ++i)
110 if (norms[i] > 0)
111 for (size_t j = 0; j < scaled.cols(); ++j)
112 scaled(i, j) /= T(norms[i]);
113 return scaled;
114}

117template <typename T>
118void matrix_symmetrize(Matrix<T>& mat) {
119 assert(mat.rows() == mat.cols() && "must be a square matrix");
120 for (size_t i = 0; i < mat.rows(); ++i)
121 for (size_t j = 0; j < i; ++j) {
122 mat(i, j) = T(0.5) * (mat(i, j) + conjugate(mat(j, i)));
123 mat(j, i) = conjugate(mat(i, j));
124 }
125}

157template <typename Slice>
158std::vector<size_t> eye_order(const Slice& mat) {
159 auto dim = mat.dimensions();
160 auto rows = std::list<size_t>{};
161 auto cols = std::list<size_t>{};
162 for (size_t i = 0; i < dim.first; ++i) {
163 rows.emplace_back(i);
164 cols.emplace_back(i);
165 }
166 auto order = std::vector<size_t>(dim.first);
167 size_t i, j;
168 while (!rows.empty() && !cols.empty()) {
169 std::tie(i, j) = max_element_index(rows, cols, mat);
170 order.at(j) = i;
171 auto it_row = std::find(begin(rows), end(rows), i);
172 auto it_col = std::find(begin(cols), end(cols), j);
173 rows.erase(it_row);
174 cols.erase(it_col);
175 }
176 return order;
177}
