50template <class R, class Q>
51auto max_overlap_with_R(const CVecRef<R>& rparams, const CVecRef<Q>& qparams, array::ArrayHandler<R, Q>& handler,
52 Logger& logger) {
53 logger.trace("max_overlap_with_R()");
54 const auto nR = rparams.size();
55 auto overlap = subspace::util::overlap(rparams, qparams, handler);
56 auto q_indices = std::vector<int>(qparams.size());
57 std::iota(std::begin(q_indices), std::end(q_indices), 0);
58 auto q_max_overlap = std::vector<int>{};
59 for (size_t i = 0; i < nR && !q_indices.empty(); ++i) {
60 auto ov = std::vector<typename array::ArrayHandler<R, Q>::value_type_abs>{};
61 for (auto j : q_indices)
62 ov.push_back(std::abs(overlap(i, j)));
63 auto it_max = std::max_element(ov.begin(), ov.end());
64 auto i_max = std::distance(ov.begin(), it_max);
65 logger.debug("removed q index = " + std::to_string(q_indices[i_max]) +
66 ", with overlap = " + std::to_string(*it_max));
67 q_max_overlap.push_back(q_indices[i_max]);
68 q_indices.erase(q_indices.begin() + i_max);
69 }
70 std::sort(std::begin(q_max_overlap), std::end(q_max_overlap), std::greater<int>());
71 return q_max_overlap;
72}

110 template <class R, class P, typename value_type, typename value_type_abs>
111 std::vector<int> run(const VecRef<R>& rparams, subspace::IXSpace<R, Q, P>& xspace,
112 const subspace::Matrix<value_type>& solutions, const value_type_abs norm_thresh,
113 const value_type_abs svd_thresh, ArrayHandlers<R, Q, P>& handlers, Logger& logger) {
114 logger.trace("DSpaceResetter::run()");
115 logger.trace("dimensions {nP, nQ, nD, nR} = ", xspace.dimensions().nP, xspace.dimensions().nQ,
116 xspace.dimensions().nD, rparams.size());
117 // solutions with columns kept in step with the changes to xspace below
118 auto solutions_xspace = solutions;
119 if (solution_params.empty() && !rparams.empty()) {
120 logger.debug("constructing solutions");
121 const auto dims = xspace.dimensions();
122 const auto overlap = xspace.data.at(subspace::EqnData::S);
123 auto q_indices = std::vector<std::size_t>(xspace.dimensions().nQ);
124 std::iota(q_indices.begin(), q_indices.end(), 0);
125 const auto solutions_proj =
126 dspace::stable_projected_solutions(solutions, overlap, dims, q_indices, norm_thresh, svd_thresh, logger);
127 const auto nC = solutions_proj.rows();
128 for (size_t i = 0; i < nC; ++i)
129 solution_params.emplace_back(util::construct_zeroed_copy(rparams.front().get(), handlers.qr()));
130 auto roots = std::vector<int>(nC);
131 std::iota(begin(roots), end(roots), 0);
132 util::construct_solutions(wrap(begin(solution_params), end(solution_params)), roots, solutions_proj, {},
133 xspace.cparamsq(), xspace.cparamsd(), 0, 0, dims.nQ, handlers.qq(), handlers.qp(),
134 handlers.qq());
135 assert(solutions_xspace.cols() == dims.nX());
136 for (size_t i = 0; i < dims.nD; ++i)
137 solutions_xspace.remove_col(dims.oD());
138 VecRef<Q> null_params, null_actions;
139 xspace.update_dspace(null_params, null_actions);
140 }
141 const auto nR = std::min(rparams.size(), solution_params.size());
142 for (size_t i = 0; i < nR; ++i) {
143 handlers.rq().copy(rparams[i], solution_params.front());
144 solution_params.pop_front();
145 }
146 const auto wparams = cwrap(begin(rparams), begin(rparams) + nR);
147 auto q_delete = max_overlap_with_R(wparams, xspace.cparamsq(), handlers.rq(), logger);
148 erase_qspace(xspace, solutions_xspace, q_delete);
149 assert(solutions_xspace.cols() == xspace.dimensions().nX());
150 if (xspace.dimensions().nQ + nR > m_max_Qsize_after_reset)
151 resize_qspace(xspace, solutions_xspace, m_max_Qsize_after_reset > nR ? m_max_Qsize_after_reset - nR : 0, logger);
152 auto new_working_set = std::vector<int>(nR);
153 std::iota(begin(new_working_set), end(new_working_set), 0);
154 return new_working_set;
155 }
156};
