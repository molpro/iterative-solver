145template <typename T>
146consteval bool is_stringifiable() {
147 using U = std::remove_cvref_t<T>;
148 if constexpr (string_like<U> || complex_number<U> || std::floating_point<U>) {
149 return true;
150 } else if constexpr (pair_like<U>) {
151 return is_stringifiable<decltype(std::declval<const U &>().first)>() &&
152 is_stringifiable<decltype(std::declval<const U &>().second)>();
153 } else if constexpr (is_printable_range<U>()) {
154 return is_stringifiable<std::ranges::range_reference_t<const U>>();
155 } else {
156 return formattable<U> || streamable<U>;
157 }
158}

165concept stringifiable = detail::is_stringifiable<T>();

176template <typename T>
177 requires stringifiable<T>
178std::string stringify(const T &t, std::size_t precision = default_precision) {
179 using U = std::remove_cvref_t<T>;
180 const bool precise = precision != default_precision;
181
182 if constexpr (string_like<U>) {
183 return std::string(std::string_view(t));
184 } else if constexpr (complex_number<U> || (std::floating_point<U> && !formattable<U>)) {
185 std::ostringstream stream;
186 if (precise)
187 stream << std::setprecision(static_cast<int>(precision));
188 stream << t;
189 return stream.str();
190 } else if constexpr (std::floating_point<U>) {
191 return precise ? std::format("{:.{}}", t, precision) : std::format("{}", t);
192 } else if constexpr (pair_like<U>) {
193 return "{ " + stringify(t.first, precision) + ": " + stringify(t.second, precision) + " }";
194 } else if constexpr (detail::is_printable_range<U>()) {
195 std::string result = "[";
196 bool first = true;
197 for (auto &&element : t) {
198 if (!first)
199 result += ", ";
200 first = false;
201 result += stringify(element, precision);
202 }
203 return result + "]";
204 } else if constexpr (formattable<U>) {
205 if (precise) {
206 try {
207 return std::vformat("{:." + std::to_string(precision) + "}", std::make_format_args(t));
208 } catch (const std::format_error &) {
209 // the type's formatter has no notion of precision
210 }
211 }
212 return std::format("{}", t);
213 } else {
214 std::ostringstream stream;
215 if (precise)
216 stream << std::setprecision(static_cast<int>(precision));
217 stream << t;
218 return stream.str();
219 }
220}

445 template <typename... Ts>
446 void msg(log::Severity severity, log::Verbosity verbosity, std::string_view message, const Ts &...args) const {
447 static_assert((log::stringifiable<Ts> && ...), "All types passed to logger must be stringify-able");
448 if (!wants(typeid(log::Generic), severity, verbosity, false))
449 return;
450 const log::Generic::Payload payload{};

452 std::string arg_str;
453 bool first = true;

455 const std::string text = log::compose(message, arg_str);
456 dispatch(log::Record::make<log::Generic>(severity, verbosity, text, payload));
457 } else {
458 dispatch(log::Record::make<log::Generic>(severity, verbosity, message, payload));
459 }

527 std::string_view msg) const {
528 bool add_colon = false;
529 switch (severity) {
530 case log::Severity::Normal:

532 case log::Severity::Warning:
533 molpro::cout << "[WARNING]";
534 add_colon = true;
535 break;
536 case log::Severity::Error:
537 molpro::cout << "[ERROR]:";
538 add_colon = true;
539 break;
540 case log::Severity::Fatal:
541 molpro::cout << "[FATAL]:";
542 add_colon = true;
543 break;
544 }
545 switch (verbosity) {
546 case log::Verbosity::None:
547 break;
548 case log::Verbosity::Info:
549 molpro::cout << "[INFO]";
550 add_colon = true;
551 break;
552 case log::Verbosity::Debug:
553 molpro::cout << "[DEBUG]";
554 add_colon = true;
555 break;
556 case log::Verbosity::Trace:
557 molpro::cout << "[TRACE]";

583 bool passes(const log::Filter &filter, log::Severity severity, log::Verbosity verbosity, bool data_dump) const {
584 return severity >= filter.min_severity.value_or(m_min_severity) &&
585 verbosity >= filter.verbosity.value_or(m_verbosity) &&
586 (!data_dump || filter.data_dumps.value_or(m_dump_data));
587 }
588
589 bool wants(const std::type_info &context, log::Severity severity, log::Verbosity verbosity, bool data_dump) const {
590 if (m_default_output && passes({}, severity, verbosity, data_dump))
591 return true;
592 for (const auto &entry : m_sinks)
593 if ((!entry.context || *entry.context == context) && passes(entry.filter, severity, verbosity, data_dump))
594 return true;
595 return false;
596 }
597
598 void dispatch(const log::Record &record) const {
599 const bool data_dump = record.is_data_dump();
600 for (const auto &entry : m_sinks)
601 if ((!entry.context || *entry.context == record.type()) &&
602 passes(entry.filter, record.severity, record.verbosity, data_dump))
603 entry.sink(record);
604 if (m_default_output && passes({}, record.severity, record.verbosity, data_dump))
605 default_message_handler(record.context, record.severity, record.verbosity, record.text());
606 }
607
609 log::Verbosity m_verbosity = log::Verbosity::Info;
611 log::Severity m_min_severity = log::Severity::Normal;
613 bool m_dump_data = false;
615 bool m_default_output = true;
617 std::vector<SinkEntry> m_sinks;
618 SinkId m_next_id = 1;
619};
