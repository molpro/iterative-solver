31// Normalises each parameter to unit length, leaving alone only vectors whose length is zero or too small to invert.
32// The threshold must not depend on the scale of the problem: a preconditioned residual can legitimately be very short
33// (for example with a large diagonal), and whether it is redundant is decided afterwards from the overlap of
34// normalised vectors.
35template <class R>
36void normalise(VecRef<R>& params, array::ArrayHandler<R, R>& handler, Logger& logger,
37 typename array::ArrayHandler<R, R>::value_type_abs thresh =
38 1 / std::numeric_limits<typename array::ArrayHandler<R, R>::value_type_abs>::max()) {
39 for (auto& p : params) {
40 // the length of a vector is a magnitude, real even when the elements are complex
41 const typename array::ArrayHandler<R, R>::value_type_abs dot = std::sqrt(std::abs(handler.dot(p, p)));
42 if (dot > thresh and std::isfinite(dot)) {
43 handler.scal(1. / dot, p);
44 } else {
45 logger.warn("parameter's length is too small for normalisation, dot = " + std::format("{:.2e}", double(dot)));
46 }
47 }
48}

84template <typename value_type>
85auto construct_projected_solution(const subspace::Matrix<value_type>& solutions, const subspace::Dimensions& dims,
86 const std::vector<std::size_t>& remove_qspace, Logger& logger) {
87 logger.trace("construct_projected_solution()");
88 const auto nQd = remove_qspace.size();
89 const auto nSol = solutions.rows();
90 const auto basis = projected_solutions_basis(dims, remove_qspace);
91 auto solutions_proj = subspace::Matrix<value_type>({nSol, basis.size()});
92 for (size_t i = 0; i < nSol; ++i)
93 for (size_t j = 0; j < basis.size(); ++j)
94 solutions_proj(i, j) = solutions(i, basis[j]);
95 logger.debug("nSol, nQd, nD", nSol, nQd, dims.nD);
96 return solutions_proj;
97}

111template <typename value_type>
112auto projected_solutions_overlap(const subspace::Matrix<value_type>& solutions_proj,
113 const subspace::Matrix<value_type>& overlap, const std::vector<std::size_t>& basis,
114 const std::vector<std::size_t>& indices) {
115 assert(solutions_proj.cols() == basis.size());
116 const auto nSol = solutions_proj.rows();
117 auto ov = subspace::Matrix<value_type>({nSol, indices.size()});
118 for (size_t i = 0; i < nSol; ++i)
119 for (size_t m = 0; m < indices.size(); ++m)
120 for (size_t k = 0; k < basis.size(); ++k)
121 ov(i, m) += conjugate(solutions_proj(i, k)) * overlap(basis[k], indices[m]);
122 return ov;
123}

136template <typename value_type>
137auto projected_solutions_overlap(const subspace::Matrix<value_type>& solutions_proj,
138 const subspace::Matrix<value_type>& overlap, const std::vector<std::size_t>& basis) {
139 const auto nSol = solutions_proj.rows();
140 const auto xu = projected_solutions_overlap(solutions_proj, overlap, basis, basis);
141 auto ov = subspace::Matrix<value_type>({nSol, nSol});
142 for (size_t i = 0; i < nSol; ++i) {
143 for (size_t j = 0; j <= i; ++j) {
144 for (size_t l = 0; l < basis.size(); ++l)
145 ov(i, j) += xu(i, l) * solutions_proj(j, l);
146 ov(j, i) = conjugate(ov(i, j));
147 }
148 }
149 return ov;
150}

177template <typename value_type, typename value_type_abs>
178void remove_null_norm_and_normalise(subspace::Matrix<value_type>& parameters, subspace::Matrix<value_type>& overlap,
179 const value_type_abs norm_thresh, Logger& logger) {
180 logger.trace("remove_null_norm_and_normalise()");
181 const auto nSol = parameters.rows();
182 auto norm_proj = std::vector<value_type_abs>(nSol, 0.);
183 for (size_t i = 0; i < nSol; ++i)
184 norm_proj[i] = std::sqrt(std::abs(overlap(i, i)));
185 for (size_t i = 0, j = 0; i < nSol; ++i) {
186 if (norm_proj[i] > norm_thresh) {
187 parameters.row(j).scal(1. / norm_proj[i]);
188 overlap.col(j).scal(1. / norm_proj[i]);
189 overlap.row(j).scal(1. / norm_proj[i]);
190 ++j;
191 } else {
192 parameters.remove_row(j);
193 overlap.remove_row_col(j, j);
194 std::stringstream ss;
195 ss << std::setprecision(3) << "remove projected solution parameter i = " << i << ", norm = " << norm_proj[i];
196 logger.info(ss.str());
197 }
198 }
199 logger.data_dump("norm = ", norm_proj);
200 logger.data_dump("parameters after normalisation = ", parameters);
201 logger.data_dump("overlap of parameters = ", overlap);
202}

216template <typename value_type, typename value_type_abs>
217auto remove_null_projected_solutions(const subspace::Matrix<value_type>& solutions_proj,
218 const subspace::Matrix<value_type>& overlap_proj, const value_type_abs svd_thresh,
219 Logger& logger) {
220 logger.trace("remove_null_projected_solutions()");
221 logger.debug("nS on entry = ", solutions_proj.rows());
222 value_type* m = const_cast<std::vector<value_type>&>(overlap_proj.data()).data();
223 auto svd_vecs = svd_system(overlap_proj.rows(), overlap_proj.cols(), array::Span(m, overlap_proj.size()),
224 std::numeric_limits<value_type_abs>::max(), true);
225 svd_vecs.remove_if([&svd_thresh](const auto& el) { return real_part(el.value) < svd_thresh; });
226 svd_vecs.sort([](const auto& lt, const auto& rt) { return real_part(lt.value) < real_part(rt.value); });
227 const auto nD = svd_vecs.size();
228 const auto nX = solutions_proj.cols();
229 auto solutions_stable = subspace::Matrix<value_type>({nD, nX});
230 auto svd = svd_vecs.begin();
231 for (size_t i = 0; i < nD; ++i, ++svd)
232 for (size_t j = 0; j < overlap_proj.cols(); ++j)
233 for (size_t k = 0; k < nX; ++k)
234 solutions_stable(i, k) += svd->v[j] * solutions_proj(j, k);
235 logger.debug("nS without null space = ", solutions_stable.rows());
236 return solutions_stable;
237}

251template <typename value_type, typename value_type_abs>
252auto stable_projected_solutions(const subspace::Matrix<value_type>& solutions,
253 const subspace::Matrix<value_type>& overlap, const subspace::Dimensions& dims,
254 const std::vector<std::size_t>& remove_qspace, const value_type_abs norm_thresh,
255 const value_type_abs svd_thresh, Logger& logger) {
256 auto solutions_proj = construct_projected_solution(solutions, dims, remove_qspace, logger);
257 auto overlap_proj = construct_projected_solutions_overlap(solutions_proj, overlap, dims, remove_qspace, logger);
258 remove_null_norm_and_normalise(solutions_proj, overlap_proj, norm_thresh, logger);
259 solutions_proj = remove_null_projected_solutions(solutions_proj, overlap_proj, svd_thresh, logger);
260 overlap_proj = construct_projected_solutions_overlap(solutions_proj, overlap, dims, remove_qspace, logger);
261 remove_null_norm_and_normalise(solutions_proj, overlap_proj, norm_thresh, logger);
262 return solutions_proj;
263}

274template <typename value_type>
275auto construct_full_subspace_overlap(const subspace::Matrix<value_type>& solutions_proj,
276 const subspace::Dimensions& dims, const std::vector<std::size_t>& remove_qspace,
277 const subspace::Matrix<value_type>& overlap, const size_t nR) {
278 const auto nDnew = solutions_proj.rows();
279 const auto nQd = remove_qspace.size();
280 const auto nQ = dims.nQ - nQd;
281 auto ov = overlap;
282 for (size_t i = 0; i < dims.nD; ++i) {
283 ov.remove_row_col(dims.oD(), dims.oD());
284 }
285 auto is_Qdelete = [&remove_qspace](size_t i) {
286 return std::find(begin(remove_qspace), end(remove_qspace), i) != end(remove_qspace);
287 };
288 for (size_t i = 0, j = 0; i < dims.nQ; ++i) {
289 if (is_Qdelete(i))
290 ov.remove_row_col(dims.oQ() + j, dims.oQ() + j);
291 else
292 ++j;
293 }
294 const auto oDnew = dims.nP + nQ + nR;
295 ov.resize({oDnew + nDnew, oDnew + nDnew});
296 // subspace indices of the remaining P+Q+R vectors, in their order in ov
297 auto remaining = std::vector<std::size_t>{};
298 for (size_t j = 0; j < dims.nP; ++j)
299 remaining.push_back(dims.oP() + j);
300 for (size_t j = 0; j < dims.nQ; ++j)
301 if (!is_Qdelete(j))
302 remaining.push_back(dims.oQ() + j);
303 for (size_t j = 0; j < nR; ++j)
304 remaining.push_back(dims.nX() + j);
305 assert(remaining.size() == oDnew);
306 const auto basis = projected_solutions_basis(dims, remove_qspace);
307 const auto ov_xv = projected_solutions_overlap(solutions_proj, overlap, basis, remaining);
308 const auto ov_xx = projected_solutions_overlap(solutions_proj, overlap, basis);
309 for (size_t i = 0; i < nDnew; ++i) {
310 for (size_t j = 0; j < oDnew; ++j) {
311 ov(oDnew + i, j) = ov_xv(i, j);
312 ov(j, oDnew + i) = conjugate(ov_xv(i, j));
313 }
314 for (size_t j = 0; j < nDnew; ++j)
315 ov(oDnew + i, oDnew + j) = ov_xx(i, j);
316 }
317 return ov;
318}

333template <class R, class Q, class P, typename value_type>
334auto append_overlap_with_r(const subspace::Matrix<value_type>& overlap, const CVecRef<R>& params,
335 const CVecRef<P>& pparams, const CVecRef<Q>& qparams, const CVecRef<Q>& dparams,
336 ArrayHandlers<R, Q, P>& handlers, Logger& logger) {
337 logger.trace("append_overlap_with_r()");
338 const auto nP = pparams.size(), nQ = qparams.size(), nD = dparams.size(), nN = params.size();
339 const auto nX = nP + nQ + nD + nN;
340 const auto oP = 0;
341 const auto oQ = oP + nP;
342 const auto oD = oQ + nQ;
343 const auto oN = oD + nD;
344 auto ov = overlap;
345 ov.resize({nX, nX}); // solutions are last
346 ov.slice({oN, oN}, {oN + nN, oN + nN}) = subspace::util::overlap(params, handlers.rr());
347 ov.slice({oN, oP}, {oN + nN, oP + nP}) = subspace::util::overlap(params, pparams, handlers.rp());
348 ov.slice({oN, oQ}, {oN + nN, oQ + nQ}) = subspace::util::overlap(params, qparams, handlers.rq());
349 ov.slice({oN, oD}, {oN + nN, oD + nD}) = subspace::util::overlap(params, dparams, handlers.rq());
350 auto copy_upper_to_lower = [&ov, oN, nN](size_t oX, size_t nX) {
351 for (size_t i = 0; i < nX; ++i)
352 for (size_t j = 0; j < nN; ++j)
353 ov(oX + i, oN + j) = ov(oN + j, oX + i);
354 };
355 copy_upper_to_lower(oP, nP);
356 copy_upper_to_lower(oQ, nQ);
357 copy_upper_to_lower(oD, nD);
358 logger.data_dump("full overlap P+Q+D+R = ", ov);
359 return ov;
360}

370template <typename value_type>
371std::vector<std::size_t> limit_qspace_size(const subspace::Dimensions& dims, const QSpaceOptions& opts,
372 const subspace::Matrix<value_type>& solutions, Logger& logger) {
373 using value_type_abs = decltype(std::abs(solutions(0, 0)));
374 using contrib_pair = std::pair<std::size_t, value_type_abs>;
375 using std::begin;
376 using std::end;
377
378 logger.trace("limit_qspace_size()");
379
380 assert(opts.max_size >= opts.min_size);
381
382 const bool must_trim_size = dims.nQ > opts.max_size;
383 const bool may_trim_size = dims.nQ > opts.min_size && opts.contrib_thresh > 0;
384
385 if (!must_trim_size && !may_trim_size) {
386 return {};
387 }
388
389 const std::size_t min_deletions = must_trim_size ? dims.nQ - opts.max_size : 0;
390 const std::size_t max_deletions = dims.nQ - opts.min_size;
391
392 std::vector<std::size_t> q_delete;
393 q_delete.reserve(min_deletions);
394
395 const std::size_t nSol = solutions.rows();
396
397 std::vector<contrib_pair> max_contrib_to_solution;
398 max_contrib_to_solution.reserve(nSol);
399
400 for (std::size_t idx : std::ranges::views::iota(std::size_t(0), dims.nQ)) {
401 max_contrib_to_solution.emplace_back(std::make_pair(idx, value_type_abs(0)));
402
403 for (size_t j = 0; j < nSol; ++j) {
404 const value_type_abs current = std::abs(solutions(j, dims.oQ() + idx));
405 max_contrib_to_solution.back().second = std::max(max_contrib_to_solution.back().second, current);
406 }
407 }
408
409 std::ranges::sort(max_contrib_to_solution, std::less<>{}, &contrib_pair::second);
410
411 logger.trace("contribution to solutions =",
412 max_contrib_to_solution | std::ranges::views::transform([](const contrib_pair& p) { return p.second; }));
413
414 for (auto [idx, contrib] : max_contrib_to_solution) {
415 if (q_delete.size() == max_deletions || (q_delete.size() >= min_deletions && contrib >= opts.contrib_thresh)) {
416 break;
417 }
418
419 if (opts.contrib_thresh != 0 && contrib >= opts.contrib_thresh) {
420 logger.warn("deleted Q vector with max. contribution = ", contrib);
421 } else {
422 logger.debug("deleted Q vector with max. contribution = ", contrib);
423 }
424
425 q_delete.emplace_back(idx);
426 }
427
428 return q_delete;
429}

442template <class R, class Q, class P, typename value_type, typename value_type_abs>
443auto construct_dspace(const subspace::Matrix<value_type>& solutions, const subspace::IXSpace<R, Q, P>& xspace,
444 const std::vector<std::size_t>& q_delete, const value_type_abs norm_thresh,
445 const value_type_abs svd_thresh, array::ArrayHandler<Q, Q>& handler, Logger& logger) {
446 const auto dims = xspace.dimensions();
447 const auto overlap = xspace.data.at(subspace::EqnData::S);
448 const auto solutions_proj =
449 dspace::stable_projected_solutions(solutions, overlap, dims, q_delete, norm_thresh, svd_thresh, logger);
450#ifndef NDEBUG
451 // a decomposition of the whole subspace overlap, wanted only by this check
452 auto overlap_full_subspace = dspace::construct_full_subspace_overlap(solutions_proj, dims, q_delete, overlap, 0);
453 auto svd_vecs = svd_system(overlap_full_subspace.rows(), overlap_full_subspace.cols(),
454 array::Span(&overlap_full_subspace(0, 0), overlap_full_subspace.size()), svd_thresh, true);
455 assert(svd_vecs.empty() && "P+Q+D subspace should be stable by construction");
456#endif
457 const auto nD = solutions_proj.rows();
458 const auto nQd = q_delete.size();
459 assert(nQd + dims.nD == solutions_proj.cols());
460 const auto& qparams = xspace.cparamsq();
461 const auto& qactions = xspace.cactionsq();
462 const auto& dparams = xspace.cparamsd();
463 const auto& dactions = xspace.cactionsd();
464 std::vector<Q> dparams_new, dactions_new;
465 if (nD > 0) {
466 const Q& q = !qparams.empty() ? qparams.front().get() : dparams.front().get();
467 for (size_t i = 0; i < nD; ++i) {
468 dparams_new.emplace_back(util::construct_zeroed_copy(q, handler));
469 dactions_new.emplace_back(util::construct_zeroed_copy(q, handler));
470 }
471 // coefficient matrix with Q_{delete} and D vectors in rows and new D vectors in columns, as required by gemm_outer
472 auto coeff = subspace::Matrix<value_type>({nQd + dims.nD, nD});
473 for (size_t i = 0; i < nD; ++i)
474 for (size_t j = 0; j < nQd + dims.nD; ++j)
475 coeff(j, i) = solutions_proj(i, j);
476 // Q_{delete} followed by D, as the columns of solutions_proj, so that one call per output passes over each new
477 // vector only once
478 auto qd_params = CVecRef<Q>{}, qd_actions = CVecRef<Q>{};
479 for (auto iq : q_delete) {
480 qd_params.emplace_back(qparams.at(iq));
481 qd_actions.emplace_back(qactions.at(iq));
482 }
483 qd_params.insert(qd_params.end(), dparams.begin(), dparams.end());
484 qd_actions.insert(qd_actions.end(), dactions.begin(), dactions.end());
485 const auto wdparams_new = wrap(dparams_new);
486 const auto wdactions_new = wrap(dactions_new);
487 handler.gemm_outer(coeff, qd_params, wdparams_new);
488 handler.gemm_outer(coeff, qd_actions, wdactions_new);
489 }
490 for (size_t i = 0; i < nD; ++i) {
491 auto norm = std::sqrt(std::abs(handler.dot(dparams_new.at(i), dparams_new.at(i))));
492 if (norm < norm_thresh) {
493 logger.warn(
494 std::format("construct_dspace: skipping normalisation of D vector {} with near-zero norm = {:.2e}", i, norm));
495 continue;
496 }
497 handler.scal(1. / norm, dparams_new[i]);
498 handler.scal(1. / norm, dactions_new[i]);
499 }
500 return std::make_tuple(std::move(dparams_new), std::move(dactions_new));
501}

523template <class R, class Q, class P, typename value_type, typename value_type_abs>
524auto project_out_and_orthonormalise(const VecRef<R>& rparams, const subspace::Matrix<value_type>& overlap,
525 const subspace::Matrix<value_type>& rx_overlap, const subspace::Dimensions& dims,
526 const CVecRef<P>& pparams, const CVecRef<Q>& qparams, const CVecRef<Q>& dparams,
527 const value_type_abs norm_thresh, ArrayHandlers<R, Q, P>& handlers,
528 Logger& logger) {
529 logger.trace("project_out_and_orthonormalise()");
530 const auto nR = rparams.size(), nP = pparams.size(), nQ = qparams.size(), nD = dparams.size();
531 assert(nP == dims.nP && nQ == dims.nQ && nD == dims.nD);
532 assert(rx_overlap.rows() == nR && rx_overlap.cols() == dims.nX());
533 // Project the new parameters out of the span of P+Q+D. The existing vectors need not be orthogonal to each other
534 // (for example user-supplied initial guesses), so subtract V S^{-1} V^dagger r rather than projecting on each vector
535 // separately.
536 auto prof = molpro::Profiler::single();
537 prof->start("orthoganalise");
538 const auto nX = nP + nQ + nD;
539 if (nR > 0 and nX > 0) {
540 using EMatrix = Eigen::Matrix<value_type, Eigen::Dynamic, Eigen::Dynamic>;
541 EMatrix S(nX, nX);
542 for (size_t i = 0; i < nX; ++i)
543 for (size_t j = 0; j < nX; ++j)
544 S(i, j) = overlap(i, j);
545 // equilibrate, so that vectors of very different lengths do not make S needlessly ill-conditioned
546 Eigen::Vector<value_type, Eigen::Dynamic> scale(nX);
547 for (size_t i = 0; i < nX; ++i)
548 scale(i) = std::abs(S(i, i)) > 0 ? 1 / std::sqrt(std::abs(S(i, i))) : 0;
549 const auto Sinv = (scale.asDiagonal() * S * scale.asDiagonal()).eval().completeOrthogonalDecomposition();
550 // the projection needs <x_i|r_j>, the conjugate of the overlaps already calculated
551 EMatrix dots(nX, nR);
552 for (size_t i = 0; i < nX; ++i)
553 for (size_t j = 0; j < nR; ++j)
554 dots(i, j) = molpro::linalg::conjugate(rx_overlap(j, i));
555 const EMatrix y = scale.asDiagonal() * Sinv.solve(scale.asDiagonal() * dots);
556 auto subtract = [&y, &rparams, nR](const auto& xparams, auto& handler, size_t o, size_t n) {
557 if (n == 0)
558 return;
559 subspace::Matrix<value_type> coeff({n, nR});
560 for (size_t i = 0; i < n; ++i)
561 for (size_t j = 0; j < nR; ++j)
562 coeff(i, j) = -y(o + i, j);
563 handler.gemm_outer(coeff, xparams, rparams);
564 };
565 subtract(pparams, handlers.rp(), dims.oP(), nP);
566 subtract(qparams, handlers.rq(), dims.oQ(), nQ);
567 subtract(dparams, handlers.rq(), dims.oD(), nD);
568 }
569 prof->stop();
570 prof->start("get null_params");
571 auto null_params = std::vector<int>{};
572 for (size_t i = 0; i < nR; ++i) {
573 auto norm = std::sqrt(std::abs(handlers.rr().dot(rparams[i], rparams[i])));
574 if (norm > norm_thresh) {
575 handlers.rr().scal(1. / norm, rparams[i]);
576 for (size_t j = i + 1; j < nR; ++j) {
577 auto ov = handlers.rr().dot(rparams[i], rparams[j]);
578 handlers.rr().axpy(-ov, rparams[i], rparams[j]);
579 }
580 } else {
581 null_params.push_back(i);
582 }
583 }
584 prof->stop();
585 return null_params;
586}

630template <class R, class Q, class P>
631auto propose_rspace(IterativeSolver<R, Q, P>& solver, const VecRef<R>& parameters, const VecRef<R>& residuals,
632 subspace::IXSpace<R, Q, P>& xspace, subspace::ISubspaceSolver<R, Q, P>& subspace_solver,
633 ArrayHandlers<R, Q, P>& handlers, Logger& logger,
634 const RSpaceOptions<typename array::ArrayHandler<R, R>::value_type_abs>& r_opts,
635 const QSpaceOptions& q_opts, molpro::profiler::Profiler& profiler) {
636 logger.trace("itsolv::detail::propose_rspace");
637 logger.trace("dimensions {nP, nQ, nD, nW} = ", xspace.dimensions().nP, xspace.dimensions().nQ, xspace.dimensions().nD,
638 solver.working_set().size());
639 profiler.start("itsolv::ISubspaceSolver::solutions");
640 auto solutions = subspace_solver.solutions();
641 profiler.stop();
642 auto q_delete = limit_qspace_size(xspace.dimensions(), q_opts, solutions, logger);
643 logger.debug("delete Q parameter indices = ", q_delete);
644 if (!q_delete.empty()) {
645 auto prof = profiler.push("construct_dspace");
646 auto [dparams, dactions] =
647 construct_dspace(solutions, xspace, q_delete, r_opts.norm_thresh, r_opts.svd_thresh, handlers.qq(), logger);
648 erase_qspace(xspace, q_delete);
649 auto wdparams = wrap(dparams);
650 auto wdactions = wrap(dactions);
651 xspace.update_dspace(wdparams, wdactions);
652 auto eigenvalues_ref = subspace_solver.eigenvalues();
653 subspace_solver.solve(xspace, solutions.rows());
654 const auto& eigenvalues_new = subspace_solver.eigenvalues();
655 const auto n_compare = std::min(eigenvalues_ref.size(), eigenvalues_new.size());
656 auto eigval_error = std::vector<double>{};
657 std::transform(std::begin(eigenvalues_ref), std::begin(eigenvalues_ref) + n_compare, std::begin(eigenvalues_new),
658 std::back_inserter(eigval_error), [](auto& e_ref, auto& e_new) { return std::abs(e_ref - e_new); });
659 logger.debug("eigenvalue error due to new D space = ", eigval_error);
660 // FIXME Optionally, solve the subspace problem again and get an estimate of the error due to new D
661 }
662 // Orthonormalise the residuals against P+Q+D and among themselves, removing redundant and null parameters.
663 auto wresidual = wrap(residuals.begin(), residuals.begin() + solver.working_set().size());
664 profiler.start("normalise");
665 normalise(wresidual, handlers.rr(), logger);
666 profiler.stop();
667 profiler.start("append_overlap_with_r");
668 const auto full_overlap =
669 append_overlap_with_r(xspace.data.at(subspace::EqnData::S), cwrap(wresidual), xspace.cparamsp(),
670 xspace.cparamsq(), xspace.cparamsd(), handlers, logger);
671 profiler.stop();
672 profiler.start("redundant_indices");
673 auto redundant_indices =
674 redundant_parameters(full_overlap, xspace.dimensions().nX(), wresidual.size(), r_opts.svd_thresh, logger);
675 profiler.stop();
676 logger.debug("redundant indices = ", redundant_indices);
677 // overlaps of the remaining R parameters with P+Q+D, to save project_out_and_orthonormalise() from
678 // recalculating them
679 const auto nX = xspace.dimensions().nX();
680 auto kept = std::vector<size_t>{};
681 for (size_t j = 0; j < wresidual.size(); ++j)
682 if (std::find(redundant_indices.begin(), redundant_indices.end(), int(j)) == redundant_indices.end())
683 kept.push_back(j);
684 auto rx_overlap = subspace::Matrix<typename decltype(full_overlap)::value_type>({kept.size(), nX});
685 for (size_t j = 0; j < kept.size(); ++j)
686 for (size_t i = 0; i < nX; ++i)
687 rx_overlap(j, i) = full_overlap(nX + kept[j], i);
688 util::delete_parameters(redundant_indices, wresidual);
689 profiler.start("project_out_and_orthonormalise");
690 auto null_param_indices = project_out_and_orthonormalise(wresidual, xspace.data.at(subspace::EqnData::S), rx_overlap,
691 xspace.dimensions(), xspace.cparamsp(), xspace.cparamsq(),
692 xspace.cparamsd(), r_opts.norm_thresh, handlers, logger);
693 profiler.stop();
694 // Now that there is SVD null_param_indices should always be empty
695 logger.debug("null parameters = ", null_param_indices);
696 util::delete_parameters(null_param_indices, wresidual);
697 normalise(wresidual, handlers.rr(), logger);
698 for (size_t i = 0; i < wresidual.size(); ++i)
699 handlers.rr().copy(parameters.at(i), wresidual.at(i));
700 profiler.start("get_new_working_set");
701 auto new_working_set = get_new_working_set(solver.working_set(), cwrap(residuals), cwrap(wresidual));
702 profiler.stop();
703 return new_working_set;
704}
