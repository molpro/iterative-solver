26inline std::vector<std::pair<size_t, size_t>> parameter_batches(const size_t nsol, const size_t nparam) {
27 auto batches = std::vector<std::pair<size_t, size_t>>{};
28 if (nparam && nsol) {
29 auto n_batch = nsol / nparam + (nsol % nparam ? 1 : 0);
30 for (size_t ib = 0, start_sol = 0, end_sol = 0; ib < n_batch; ++ib, start_sol = end_sol) {
31 end_sol = std::min(start_sol + nparam, nsol);
32 batches.emplace_back(start_sol, end_sol);
33 }
34 }
35 return batches;

38template <class R, class Q, class P, typename value_type>
39void construct_solution(const VecRef<R>& params, const std::vector<int>& roots,
40 const subspace::Matrix<value_type>& solutions,
41 const std::vector<std::reference_wrapper<P>>& pparams,
42 const std::vector<std::reference_wrapper<Q>>& qparams,
43 const std::vector<std::reference_wrapper<Q>>& dparams, size_t oP, size_t oQ, size_t oD,
44 ArrayHandlers<R, Q, P>& handlers) {
45 if (roots.empty())
46 return;
47 auto prof = molpro::Profiler::single();
48 prof->start("get rd_mat");
49 assert(params.size() >= roots.size());
50 for (size_t i = 0; i < roots.size(); ++i) {
51 handlers.rr().fill(0, params.at(i));
52 }
53 subspace::Matrix<value_type> rp_mat(std::make_pair(pparams.size(), roots.size())),
54 rq_mat(std::make_pair(qparams.size(), roots.size())), rd_mat(std::make_pair(dparams.size(), roots.size()));
55 for (size_t i = 0; i < roots.size(); ++i) {
56 for (size_t j = 0; j < pparams.size(); ++j) {
57 rp_mat(j, i) = solutions(roots[i], oP + j);
58 }
59 for (size_t j = 0; j < qparams.size(); ++j) {
60 rq_mat(j, i) = solutions(roots[i], oQ + j);
61 }
62 for (size_t j = 0; j < dparams.size(); ++j) {
63 rd_mat(j, i) = solutions(roots[i], oD + j);
64 }
65 }
66 prof->stop();
67 handlers.rp().gemm_outer(rp_mat, cwrap(pparams), params);
68 handlers.rq().gemm_outer(rq_mat, cwrap(qparams), params);
69 handlers.rq().gemm_outer(rd_mat, cwrap(dparams), params);
70}

101template <typename T>
102std::vector<int> select_working_set(const size_t nw, const std::vector<T>& errors, const T threshold,
103 const std::vector<T>& value_errors, const T value_threshold) {
104 auto ordered_errors = std::multimap<T, size_t, std::greater<T>>{};
105 for (size_t i = 0; i < errors.size(); ++i) {
106 if (errors[i] > threshold or (i < value_errors.size() and value_errors[i] > value_threshold))
107 ordered_errors.emplace(errors[i], i);
108 }
109 auto working_set = std::vector<int>{};
110 auto end = (ordered_errors.size() < nw ? ordered_errors.end() : next(begin(ordered_errors), nw));
111 std::transform(begin(ordered_errors), end, std::back_inserter(working_set), [](const auto& el) { return el.second; });
112 std::sort(working_set.begin(), working_set.end());
113 return working_set;
114}

426 bool test_problem(const Problem<R>& problem, R& v0, R& v1, int verbosity, value_type_abs threshold) const override {
427 bool success = true;
428 value_type_abs step = 1e-4;
429 if (this->nonlinear()) {
430 if (!problem.test_parameters(0, v0))
431 return true;
432 auto value0 = problem.residual(v0, v1);
433 if (verbosity > 1) {
434 std::cout << "value0 " << value0 << std::endl;
435 // std::cout << "parameters0 " << v0[0] << "," << v0[1] << ",..." << std::endl;
436 // std::cout << "residual0 " << v1[0] << "," << v1[1] << ",..." << std::endl;
437 }
438 Q parameters0 = m_handlers->qr().copy(v0);
439 Q residual0 = m_handlers->qr().copy(v1);
440 for (int instance = 1; problem.test_parameters(instance, v0); ++instance) {
441 m_handlers->rq().axpy(-1.0, parameters0, v0);
442 m_handlers->rr().scal(1 / std::sqrt(std::abs(m_handlers->rr().dot(v0, v0))), v0);
443 Q step1 = m_handlers->qr().copy(v0);
444 auto residual_analytic = m_handlers->rq().dot(v0, residual0);
445 m_handlers->rq().copy(v0, parameters0);
446 m_handlers->rq().axpy(-2*step, step1, v0);
447 auto valuem2 = problem.residual(v0, v1);
448 m_handlers->rq().copy(v0, parameters0);
449 m_handlers->rq().axpy(-1*step, step1, v0);
450 auto valuem1 = problem.residual(v0, v1);
451 m_handlers->rq().copy(v0, parameters0);
452 m_handlers->rq().axpy(+1*step, step1, v0);
453 auto valuep1 = problem.residual(v0, v1);
454 m_handlers->rq().copy(v0, parameters0);
455 m_handlers->rq().axpy(+2*step, step1, v0);
456 auto valuep2 = problem.residual(v0, v1);
457 auto residual_numerical = (valuem2 - value_type(8) * valuem1 + value_type(8) * valuep1 - valuep2) /
458 value_type(12 * step);
459 if (verbosity > 1)
460 std::cout << "testing problem class, instance: " << instance << ", numerical: " << residual_numerical <<", analytical: "<<residual_analytic <<", difference: "<<residual_numerical-residual_analytic << std::endl;
461 success = success && std::abs(residual_numerical - residual_analytic) < threshold;
462 }
463 } else {
464 // checks that the operator applied by apply(x, y) scales linearly; name describes its result
465 auto check_scaling = [&](const auto& apply, const std::string& name) {
466 apply(v0, v1);
467 Q residual = m_handlers->qr().copy(v1);
468 auto norm2_residual = std::sqrt(m_handlers->rr().dot(v1, v1));
469 const value_type_abs scale_factor{10.0};
470 m_handlers->rr().scal(scale_factor, v0);
471 apply(v0, v1);
472 m_handlers->rq().axpy(-scale_factor, residual, v1);
473 auto norm2 = std::sqrt(m_handlers->rr().dot(v1, v1));
474 success = success && std::abs(norm2 / norm2_residual) < threshold;
475 if (verbosity > 0 or (verbosity > -1 and not success))
476 std::cout << "Length of " << name << ": " << norm2_residual << ", scaling defect: " << norm2 << std::endl;
477 };
478 for (int instance = 0; problem.test_parameters(instance, v0); ++instance) {
479 check_scaling([&problem](R& x, R& y) { problem.action({x}, {y}); }, "residual");
480 if (problem.metric_kind() != MetricKind::Identity)
481 check_scaling([&problem](R& x, R& y) { problem.metric_action({x}, {y}); }, "action of the metric");
482 }
483 if (problem.metric_kind() != MetricKind::Identity)
484 success = test_metric(problem, v0, v1, verbosity, threshold) && success;
485 }
486 return success;
487 }

499 bool solve_impl(const VecRef<R>& parameters, const VecRef<R>& actions, const VecRef<R>& metric_actions,
500 const Problem<R>& problem, bool generate_initial_guess) {
501 if (parameters.empty())
502 throw std::runtime_error("Empty container passed to IterativeSolver::solve()");
503 if (parameters.size() != actions.size())
504 throw std::runtime_error("Inconsistent container sizes in IterativeSolver::solve()");
505 // applies the verbosity of the solver to its logger, which repeating does not change
506 if (this->m_verbosity == Verbosity::None) {
507 // Backwards compatibility
508 this->m_logger->set_verbosity(log::Verbosity::None);
509 }
510 if (this->m_verbosity == Verbosity::Detailed) {
511 this->m_logger->set_verbosity(log::Verbosity::Trace);
512 this->m_logger->enable_data_dumps(true);
513 }
514 const auto kind = problem.metric_kind();
515 const bool metric = kind != MetricKind::Identity;
516 const bool store_metric_actions = metric and problem.metric_storage() == MetricStorage::Stored;
517 check_metric_change(kind, store_metric_actions ? MetricStorage::Stored : MetricStorage::OnTheFly);
518 // the problem is checked before the solver changes, using the buffers as scratch space
519 bool use_diagonals = problem.diagonals(actions.at(0));
520 if (generate_initial_guess and not use_diagonals)
521 throw std::runtime_error("Default initial guess requested, but diagonal elements are not available");
522 std::unique_ptr<Q> diagonals;
523 if (use_diagonals)
524 diagonals.reset(new Q{m_handlers->qr().copy(actions.at(0))});
525 std::unique_ptr<Q> metric_diagonals;
526 if (metric and use_diagonals) {
527 if (not problem.metric_diagonals(actions.at(0)))
528 throw std::invalid_argument("The problem provides diagonals() but not metric_diagonals(), without which the "
529 "default preconditioner would ignore the metric");
530 metric_diagonals.reset(new Q{m_handlers->qr().copy(actions.at(0))});
531 }
532 // the initial guess and the P space are selected from the lowest of these values
533 const Q* selection = diagonals.get();
534 std::unique_ptr<Q> ratios;
535 if (metric_diagonals and (generate_initial_guess or m_max_p > 0)) {
536 // the lowest eigenvalues of a generalised eigenproblem are expected at the lowest ratios of the diagonals, which
537 // are exact, as a regulariser would make them depend on the scale of the metric
538 m_handlers->rq().copy(parameters.at(0), *diagonals);
539 detail::precondition_default(wrap_arg(parameters.at(0).get()), std::vector<value_type>{0}, actions.at(0).get(),
540 static_cast<const R*>(nullptr), false);
541 ratios.reset(new Q{m_handlers->qr().copy(parameters.at(0))});
542 selection = ratios.get();
543 }
544 std::vector<P> pspace;
545 if (use_diagonals and m_max_p > 0)
546 pspace = select_pspace(*selection);
547 auto pspace_matrices = pspace_problem_matrices(pspace, problem, metric, store_metric_actions);
548 prepare_solve(problem, actions);
549 if (metric and not store_metric_actions)
550 set_metric(kind, MetricStorage::OnTheFly,
551 [&problem](const CVecRef<R>& parameters, const VecRef<R>& metric_actions) {
552 problem.metric_action(parameters, metric_actions);
553 });
554 else
555 set_metric(kind);
556 int nwork = parameters.size();
557 // a P space replaces any initial guess; without one, max_p > 0 still means the contents of parameters are not used
558 if (pspace.empty() and (generate_initial_guess or (use_diagonals and m_max_p > 0)))
559 generate_initial_guess_from_diagonals(parameters, *selection);
560 // the ratios of the diagonal elements are not needed any more
561 selection = nullptr;
562 ratios.reset();
563 if (!pspace.empty())
564 nwork = initialise_pspace(pspace, problem, pspace_matrices, parameters, actions);
565 for (auto iter = 0; iter < this->m_max_iter && nwork > 0; iter++) {
566 // the logging context carries the errors as doubles: a log line only needs their magnitude
567 m_logger->template info<log::NewIteration>("Iteration, Current errors", iter,
568 std::vector<double>(m_errors.begin(), m_errors.end()));
569 value_type value;
570 if (this->nonlinear()) {
571 value = problem.residual(*parameters.begin(), *actions.begin());
572 nwork = this->add_vector(*parameters.begin(), *actions.begin(), value);
573 } else if (iter > 0 or pspace.empty()) {
574 // add_vector() takes up no more vectors than there are in the working set
575 const auto naction = std::min(size_t(nwork), m_working_set.size());
576 const auto wparameters = cwrap(parameters.begin(), parameters.begin() + naction);
577 const auto wactions = wrap(actions.begin(), actions.begin() + naction);
578 if (naction > 0 and metric)
579 problem.action_with_metric(wparameters, wactions,
580 wrap(metric_actions.begin(), metric_actions.begin() + naction));
581 else if (naction > 0)
582 problem.action(wparameters, wactions);
583 nwork = metric ? this->add_vector(parameters, actions, metric_actions) : this->add_vector(parameters, actions);
584 }
585 while (this->end_iteration_needed()) {
586 if (nwork > 0) {
587 if (metric_diagonals) {
588 // parameters[0] and metric_actions[0] are overwritten to hand the diagonals to the preconditioner
589 m_handlers->rq().copy(parameters.at(0), *diagonals);
590 m_handlers->rq().copy(metric_actions.at(0), *metric_diagonals);
591 problem.precondition_with_metric(wrap(actions.begin(), actions.begin() + nwork),
592 this->working_set_eigenvalues(), parameters.at(0), metric_actions.at(0));
593 } else if (use_diagonals) {
594 // parameters[0] is overwritten to hand the diagonals to the preconditioner
595 m_handlers->rq().copy(parameters.at(0), *diagonals);
596 problem.precondition(wrap(actions.begin(), actions.begin() + nwork), this->working_set_eigenvalues(),
597 parameters.at(0));
598 } else
599 problem.precondition(wrap(actions.begin(), actions.begin() + nwork), this->working_set_eigenvalues());
600 }
601 nwork = this->end_iteration(parameters, actions);
602 }
603 if (verbosity_at_least(Verbosity::Iteration)) {
604 iteration_report();
605 }
606 }
607
608 this->finalize();
609
610 if (verbosity_at_least(Verbosity::Summary)) {
611 summary_report();
612 }
613 // without any errors (e.g. no iteration was done) nothing is known to have converged
614 const bool errors_converged =
615 !m_errors.empty() && *std::max_element(m_errors.begin(), m_errors.end()) <= m_convergence_threshold;
616 // roots also remain in the working set while their values change by more than the value threshold
617 const bool converged = nwork == 0 and errors_converged;
618 if (verbosity_at_least(Verbosity::Summary)) {
619 if (!errors_converged) {
620 this->m_logger->warn("Solver has not converged to threshold ", m_convergence_threshold);
621 } else if (!converged) {
622 this->m_logger->warn("Solver has not converged to value threshold ", m_convergence_threshold_value);
623 } else {
624 this->m_logger->info("Solver converged");
625 }
626 }
627 return converged;
628 }

635 void solution_impl(const std::vector<int>& roots, const VecRef<R>& parameters, const VecRef<R>& residual,
636 const VecRef<R>& scratch) {
637 // auto prof = profiler()->push("itsolv::solution"); // FIXME two profilers
638 auto prof = molpro::Profiler::single();
639 check_consistent_number_of_roots_and_solutions(roots, parameters.size());
640 prof->start("construct_solution (parameters)");
641 detail::construct_solution(parameters, roots, m_subspace_solver->solutions(), m_xspace->paramsp(),
642 m_xspace->paramsq(), m_xspace->paramsd(), m_xspace->dimensions().oP(),
643 m_xspace->dimensions().oQ(), m_xspace->dimensions().oD(), *m_handlers);
644 prof->stop();
645 auto metric_solutions = detail::MetricSolutions<R, Q>{};
646 auto metric_solutions_q = std::vector<Q>{};
647 if (m_metric_kind != MetricKind::Identity && m_metric_storage == MetricStorage::OnTheFly && !roots.empty()) {
648 prof->start("apply metric");
649 const auto solutions = cwrap(parameters.begin(), parameters.begin() + roots.size());
650 const bool use_scratch = scratch.size() >= roots.size();
651 // without scratch vectors, the residuals hold the actions of the metric until they are copied
652 const auto metric_actions = use_scratch ? wrap(scratch.begin(), scratch.begin() + roots.size())
653 : wrap(residual.begin(), residual.begin() + roots.size());
654 m_apply_metric(solutions, metric_actions);
655 if (use_scratch) {
656 metric_solutions.r = cwrap(metric_actions);
657 } else {
658 for (const auto& m : metric_actions)
659 metric_solutions_q.emplace_back(m_handlers->qr().copy(m));
660 m_stats->q_creations += metric_solutions_q.size();
661 metric_solutions.q = cwrap(metric_solutions_q);
662 }
663 prof->stop();
664 }
665 prof->start("construct_solution (residual)");
666 detail::construct_solution(residual, roots, m_subspace_solver->solutions(), {}, m_xspace->actionsq(),
667 m_xspace->actionsd(), m_xspace->dimensions().oP(), m_xspace->dimensions().oQ(),
668 m_xspace->dimensions().oD(), *m_handlers);
669 prof->stop();
670 prof->start("apply");
671 auto pvectors = detail::construct_vectorP(roots, m_subspace_solver->solutions(), m_xspace->dimensions().oP(),
672 m_xspace->dimensions().nP);
673 if (m_apply_p)
674 m_apply_p(pvectors, m_xspace->cparamsp(), residual);
675 construct_residual(roots, cwrap(parameters), residual, metric_solutions);
676 read_handler_counts(m_stats, m_handlers);
677 prof->stop();
678 }

720 bool test_metric(const Problem<R>& problem, R& v0, R& v1, int verbosity, value_type_abs threshold) const {
721 bool success = true;
722 const bool positive = problem.metric_kind() == MetricKind::HermitianPositiveDefinite ||
723 problem.metric_kind() == MetricKind::HermitianPositiveSemidefinite;
724 std::optional<std::pair<Q, Q>> u; // the first test parameter and its action of the metric
725 for (int instance = 0; instance < 2 && problem.test_parameters(instance, v0); ++instance) {
726 problem.metric_action({v0}, {v1});
727 const auto norm = std::sqrt(std::abs(m_handlers->rr().dot(v0, v0)));
728 const auto norm_metric_action = std::sqrt(std::abs(m_handlers->rr().dot(v1, v1)));
729 const auto vmv = m_handlers->rr().dot(v0, v1);
730 if (std::abs(vmv - conjugate(vmv)) / 2 > threshold * norm * norm_metric_action or
731 (positive and real_part(vmv) < -threshold * norm * norm_metric_action)) {
732 success = false;
733 if (verbosity > -1)
734 std::cout << "<v|M v> = " << vmv << " is not real" << (positive ? " and non-negative" : "") << std::endl;
735 }
736 if (u) {
737 const auto umv = conjugate(m_handlers->rq().dot(v1, u->first));
738 const auto muv = m_handlers->rq().dot(v0, u->second);
739 const auto scale = std::sqrt(std::abs(m_handlers->qq().dot(u->first, u->first)) *
740 std::abs(m_handlers->qq().dot(u->second, u->second))) *
741 norm * norm_metric_action;
742 const auto defect = std::abs(umv - conjugate(muv));
743 if (defect > threshold * std::sqrt(scale)) {
744 success = false;
745 if (verbosity > -1)
746 std::cout << "The metric is not Hermitian: <u|M v> = " << umv << ", <M u|v> = " << conjugate(muv)
747 << std::endl;
748 } else if (verbosity > 0)
749 std::cout << "Hermiticity defect of the metric: " << defect << std::endl;
750 } else
751 u.emplace(m_handlers->qr().copy(v0), m_handlers->qr().copy(v1));
752 }
753 return success;
754 }

875 check_metric_positive();

877 // the actions of the metric are not needed any more and serve as scratch space

879 prof->stop();

892 size_t solve_and_generate_working_set(const VecRef<R>& parameters, const VecRef<R>& action,
893 const VecRef<R>& scratch = {}) {
894 auto prof = profiler();
895 auto p = prof->push("itsolv::solve_and_generate_working_set");
896 prof->start("itsolv::solve");
897 m_subspace_solver->solve(*m_xspace, n_roots());
898 prof->stop();
899 prof->start("itsolv::temp_solutions");
900 auto nsol = m_subspace_solver->size();
901 std::vector<std::pair<Q, Q>> temp_solutions{};
902 const auto batches = detail::parameter_batches(nsol, parameters.size());
903 // The solutions of a problem with a metric are normalised in the metric, so their Euclidean length, and with it
904 // the length of the residual, would depend on the scale of the metric. The error is that of the solution of unit
905 // length instead, which is the same as without a metric.
906 const auto solution_norms =
907 m_xspace->metric_kind() != MetricKind::Identity
908 ? subspace::util::solution_norms(m_subspace_solver->solutions(), m_xspace->data.at(subspace::EqnData::S))
909 : std::vector<value_type_abs>{};
910 for (const auto& batch : batches) {
911 auto [start_sol, end_sol] = batch;
912 auto roots = std::vector<int>(end_sol - start_sol);
913 std::iota(begin(roots), end(roots), start_sol);
914 solution_impl(roots, parameters, action, scratch);
915 auto errors = std::vector<value_type_abs>(roots.size(), 0);
916 compute_errors(roots, cwrap(parameters), cwrap(action), solution_norms, errors);
917 if (batches.size() > 1) {
918 for (size_t i = 0; i < roots.size(); ++i)
919 temp_solutions.emplace_back(m_handlers->qr().copy(parameters[i]), m_handlers->qr().copy(action[i]));
920 m_stats->q_creations += 2 * roots.size();
921 }
922 m_subspace_solver->set_error(roots, errors);
923 }
924 prof->stop();
925 combine_errors();
926 set_value_errors();
927 m_errors = m_subspace_solver->errors();
928 m_working_set = detail::select_working_set(parameters.size(), m_errors, m_convergence_threshold, m_value_errors,
929 m_convergence_threshold_value);
930 for (size_t i = 0; i < m_working_set.size(); ++i) {
931 size_t root = m_working_set[i];
932 if (batches.size() > 1) {
933 m_handlers->rq().copy(parameters[i], temp_solutions.at(root).first);
934 m_handlers->rq().copy(action[i], temp_solutions.at(root).second);
935 } else {
936 if (root < i)
937 throw std::logic_error("incorrect ordering of roots");
938 if (root > i && root < parameters.size()) {
939 m_handlers->rr().copy(parameters[i], parameters[root]);
940 m_handlers->rr().copy(action[i], action[root]);
941 }
942 }
943 }
944 m_logger->trace("add_vector::errors = ", m_errors);
945 return m_working_set.size();
946 }

996 std::vector<P> select_pspace(const Q& diagonals) {
997 auto selectp = m_handlers->qq().select(m_max_p, diagonals);
998 if (!selectp.empty()) {
999 // selectp is keyed by index, not value; find the smallest selected
1000 // diagonal explicitly before applying the threshold.
1001 // the diagonal of a hermitian operator is real, so it is orderable
1002 auto min_val = real_part(std::min_element(selectp.begin(), selectp.end(),
1003 [](const auto& a, const auto& b) {
1004 return real_part(a.second) < real_part(b.second);
1005 })
1006 ->second);
1007 for (auto s = selectp.begin(); s != selectp.end();) {
1008 if (real_part(s->second) > min_val + m_p_threshold)
1009 s = selectp.erase(s);
1010 else
1011 ++s;
1012 }
1013 }
1014 if (!selectp.empty() && selectp.size() < n_roots()) {
1015 this->m_logger->warn("P space smaller than number of roots is not used, starting from the diagonal elements "
1016 "instead; dimension and number of roots",
1017 selectp.size(), n_roots());
1018 selectp.clear();
1019 }
1020 if (!selectp.empty() && verbosity_at_least(Verbosity::Summary)) {
1021 this->m_logger->info("P-space dimension, threshold and limit", selectp.size(), m_p_threshold, m_max_p);
1022 }
1023 if (verbosity_at_least(Verbosity::Detailed)) {
1024 for (const auto& s : selectp) {
1025 this->m_logger->debug(std::format("P space element {}: {:.6e}", s.first, double(real_part(s.second))));
1026 }
1027 }
1028 std::vector<P> pspace;
1029 for (const auto& s : selectp)
1030 pspace.emplace_back((P){{s.first, 1}});
1031 return pspace;
1032 }
