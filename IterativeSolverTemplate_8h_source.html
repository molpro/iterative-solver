26inline std::vector<std::pair<size_t, size_t>> parameter_batches(const size_t nsol, const size_t nparam) {
27 auto batches = std::vector<std::pair<size_t, size_t>>{};
28 if (nparam && nsol) {
29 auto n_batch = nsol / nparam + (nsol % nparam ? 1 : 0);
30 for (size_t ib = 0, start_sol = 0, end_sol = 0; ib < n_batch; ++ib, start_sol = end_sol) {
31 end_sol = std::min(start_sol + nparam, nsol);
32 batches.emplace_back(start_sol, end_sol);
33 }
34 }
35 return batches;
36}

65template <typename T>
66std::vector<int> select_working_set(const size_t nw, const std::vector<T>& errors, const T threshold,
67 const std::vector<T>& value_errors, const T value_threshold) {
68 auto ordered_errors = std::multimap<T, size_t, std::greater<T>>{};
69 for (size_t i = 0; i < errors.size(); ++i) {
70 if (errors[i] > threshold or (i < value_errors.size() and value_errors[i] > value_threshold))
71 ordered_errors.emplace(errors[i], i);
72 }
73 auto working_set = std::vector<int>{};
74 auto end = (ordered_errors.size() < nw ? ordered_errors.end() : next(begin(ordered_errors), nw));
75 std::transform(begin(ordered_errors), end, std::back_inserter(working_set), [](const auto& el) { return el.second; });
76 std::sort(working_set.begin(), working_set.end());
77 return working_set;
78}

469 bool test_problem(const Problem<R>& problem, R& v0, R& v1, int verbosity, value_type_abs threshold) const override {
470 bool success = true;
471 value_type_abs step = 1e-4;
472 if (this->nonlinear()) {
473 if (!problem.test_parameters(0, v0))
474 return true;
475 auto value0 = problem.residual(v0, v1);
476 if (verbosity > 1) {
477 std::cout << "value0 " << value0 << std::endl;
478 // std::cout << "parameters0 " << v0[0] << "," << v0[1] << ",..." << std::endl;
479 // std::cout << "residual0 " << v1[0] << "," << v1[1] << ",..." << std::endl;
480 }
481 Q parameters0 = m_handlers->qr().copy(v0);
482 Q residual0 = m_handlers->qr().copy(v1);
483 for (int instance = 1; problem.test_parameters(instance, v0); ++instance) {
484 m_handlers->rq().axpy(-1.0, parameters0, v0);
485 m_handlers->rr().scal(1 / util::norm(m_handlers->rr(), v0), v0);
486 Q step1 = m_handlers->qr().copy(v0);
487 auto residual_analytic = m_handlers->rq().dot(v0, residual0);
488 m_handlers->rq().copy(v0, parameters0);
489 m_handlers->rq().axpy(-2*step, step1, v0);
490 auto valuem2 = problem.residual(v0, v1);
491 m_handlers->rq().copy(v0, parameters0);
492 m_handlers->rq().axpy(-1*step, step1, v0);
493 auto valuem1 = problem.residual(v0, v1);
494 m_handlers->rq().copy(v0, parameters0);
495 m_handlers->rq().axpy(+1*step, step1, v0);
496 auto valuep1 = problem.residual(v0, v1);
497 m_handlers->rq().copy(v0, parameters0);
498 m_handlers->rq().axpy(+2*step, step1, v0);
499 auto valuep2 = problem.residual(v0, v1);
500 auto residual_numerical = (valuem2 - value_type(8) * valuem1 + value_type(8) * valuep1 - valuep2) /
501 value_type(12 * step);
502 if (verbosity > 1)
503 std::cout << "testing problem class, instance: " << instance << ", numerical: " << residual_numerical <<", analytical: "<<residual_analytic <<", difference: "<<residual_numerical-residual_analytic << std::endl;
504 success = success && std::abs(residual_numerical - residual_analytic) < threshold;
505 }
506 } else {
507 // checks that the operator applied by apply(x, y) scales linearly; name describes its result
508 auto check_scaling = [&](const auto& apply, const std::string& name) {
509 apply(v0, v1);
510 Q residual = m_handlers->qr().copy(v1);
511 auto norm2_residual = util::norm(m_handlers->rr(), v1);
512 const value_type_abs scale_factor{10.0};
513 m_handlers->rr().scal(scale_factor, v0);
514 apply(v0, v1);
515 m_handlers->rq().axpy(-scale_factor, residual, v1);
516 auto norm2 = util::norm(m_handlers->rr(), v1);
517 success = success && std::abs(norm2 / norm2_residual) < threshold;
518 if (verbosity > 0 or (verbosity > -1 and not success))
519 std::cout << "Length of " << name << ": " << norm2_residual << ", scaling defect: " << norm2 << std::endl;
520 };
521 for (int instance = 0; problem.test_parameters(instance, v0); ++instance) {
522 check_scaling([&problem](R& x, R& y) { problem.action({x}, {y}); }, "residual");
523 if (problem.metric_kind() != MetricKind::Identity)
524 check_scaling([&problem](R& x, R& y) { problem.metric_action({x}, {y}); }, "action of the metric");
525 }
526 if (problem.metric_kind() != MetricKind::Identity)
527 success = test_metric(problem, v0, v1, verbosity, threshold) && success;
528 }
529 return success;
530 }

542 bool solve_impl(const VecRef<R>& parameters, const VecRef<R>& actions, const VecRef<R>& metric_actions,
543 const Problem<R>& problem, bool generate_initial_guess) {
544 if (parameters.empty())
545 throw std::runtime_error("Empty container passed to IterativeSolver::solve()");
546 if (parameters.size() != actions.size())
547 throw std::runtime_error("Inconsistent container sizes in IterativeSolver::solve()");
548 // applies the verbosity of the solver to its logger, which repeating does not change
549 if (this->m_verbosity == Verbosity::None) {
550 // Backwards compatibility
551 this->m_logger->set_verbosity(log::Verbosity::None);
552 }
553 if (this->m_verbosity == Verbosity::Detailed) {
554 this->m_logger->set_verbosity(log::Verbosity::Trace);
555 this->m_logger->enable_data_dumps(true);
556 }
557 const auto kind = problem.metric_kind();
558 const bool metric = kind != MetricKind::Identity;
559 const bool store_metric_actions = metric and problem.metric_storage() == MetricStorage::Stored;
560 check_metric_change(kind, store_metric_actions ? MetricStorage::Stored : MetricStorage::OnTheFly);
561 // the problem is checked before the solver changes, using the buffers as scratch space
562 bool use_diagonals = problem.diagonals(actions.at(0));
563 if (generate_initial_guess and not use_diagonals)
564 throw std::runtime_error("Default initial guess requested, but diagonal elements are not available");
565 std::unique_ptr<Q> diagonals;
566 if (use_diagonals)
567 diagonals.reset(new Q{m_handlers->qr().copy(actions.at(0))});
568 std::unique_ptr<Q> metric_diagonals;
569 if (metric and use_diagonals) {
570 if (not problem.metric_diagonals(actions.at(0)))
571 throw std::invalid_argument("The problem provides diagonals() but not metric_diagonals(), without which the "
572 "default preconditioner would ignore the metric");
573 metric_diagonals.reset(new Q{m_handlers->qr().copy(actions.at(0))});
574 }
575 // the initial guess and the P space are selected from the lowest of these values
576 const Q* selection = diagonals.get();
577 std::unique_ptr<Q> ratios;
578 if (metric_diagonals and (generate_initial_guess or m_max_p > 0)) {
579 // the lowest eigenvalues of a generalised eigenproblem are expected at the lowest ratios of the diagonals, which
580 // are exact, as a regulariser would make them depend on the scale of the metric
581 m_handlers->rq().copy(parameters.at(0), *diagonals);
582 detail::precondition_default(wrap_arg(parameters.at(0).get()), std::vector<value_type>{0}, actions.at(0).get(),
583 static_cast<const R*>(nullptr), false);
584 ratios.reset(new Q{m_handlers->qr().copy(parameters.at(0))});
585 selection = ratios.get();
586 }
587 std::vector<P> pspace;
588 if (use_diagonals and m_max_p > 0)
589 pspace = select_pspace(*selection);
590 auto pspace_matrices = pspace_problem_matrices(pspace, problem, metric, store_metric_actions);
591 prepare_solve(problem, actions);
592 if (metric and not store_metric_actions)
593 set_metric(kind, MetricStorage::OnTheFly,
594 [&problem](const CVecRef<R>& parameters, const VecRef<R>& metric_actions) {
595 problem.metric_action(parameters, metric_actions);
596 });
597 else
598 set_metric(kind);
599 int nwork = parameters.size();
600 // a P space replaces any initial guess; without one, max_p > 0 still means the contents of parameters are not used
601 const bool initial_guess =
602 pspace.empty() and (generate_initial_guess or (use_diagonals and m_max_p > 0));
603 if (initial_guess)
604 generate_initial_guess_from_diagonals(parameters, *selection);
605 // the ratios of the diagonal elements are not needed any more
606 selection = nullptr;
607 ratios.reset();
608 if (!pspace.empty())
609 nwork = initialise_pspace(pspace, problem, pspace_matrices, parameters, actions);
610 m_logger->template emit<log::SolveStarted<scalar_type>>(log::Severity::Normal, log::Verbosity::Debug, "Solve started",
611 [&] {
612 return typename log::SolveStarted<scalar_type>::Payload{
613 n_roots(), this->m_max_iter, m_convergence_threshold,
614 pspace.size(), initial_guess};
615 });
616 // the roots worked on in the current iteration, only recorded if the iteration events are wanted
617 std::vector<int> roots;
618 const bool record_roots =
619 m_logger->template is_wanted<log::NewIteration<scalar_type>>(log::Severity::Normal, log::Verbosity::Debug) ||
620 m_logger->template is_wanted<log::IterationCompleted<scalar_type>>(log::Severity::Normal, log::Verbosity::Info);
621 int iter = 0;
622 // an iteration that resets the D space is followed by one that takes up its vectors, even beyond max_iter, as
623 // there are no solutions to return in between
624 for (; (iter < this->m_max_iter || m_solutions_outdated) && nwork > 0; iter++) {
625 // the vectors worked on in this iteration: add_vector() takes up no more vectors than there are in the working
626 // set, and in the first iteration after initialising a P space the actions are already known
627 std::size_t naction = 0;
628 if (this->nonlinear())
629 naction = std::min(std::size_t(1), m_working_set.size());
630 else if (iter > 0 or pspace.empty())
631 naction = std::min(size_t(nwork), m_working_set.size());
632 // add_vector() replaces the working set, so it is recorded beforehand
633 if (record_roots)
634 roots.assign(m_working_set.begin(), m_working_set.begin() + naction);
635 m_logger->template emit<log::NewIteration<scalar_type>>(
636 log::Severity::Normal, log::Verbosity::Debug, "Iteration ",
637 [&] { return typename log::NewIteration<scalar_type>::Payload{iter, roots}; });
638 value_type value;
639 if (this->nonlinear()) {
640 value = problem.residual(*parameters.begin(), *actions.begin());
641 nwork = this->add_vector(*parameters.begin(), *actions.begin(), value);
642 } else if (iter > 0 or pspace.empty()) {
643 const auto wparameters = cwrap(parameters.begin(), parameters.begin() + naction);
644 const auto wactions = wrap(actions.begin(), actions.begin() + naction);
645 if (naction > 0 and metric)
646 problem.action_with_metric(wparameters, wactions,
647 wrap(metric_actions.begin(), metric_actions.begin() + naction));
648 else if (naction > 0)
649 problem.action(wparameters, wactions);
650 nwork = metric ? this->add_vector(parameters, actions, metric_actions) : this->add_vector(parameters, actions);
651 }
652 while (this->end_iteration_needed()) {
653 if (nwork > 0) {
654 if (metric_diagonals) {
655 // parameters[0] and metric_actions[0] are overwritten to hand the diagonals to the preconditioner
656 m_handlers->rq().copy(parameters.at(0), *diagonals);
657 m_handlers->rq().copy(metric_actions.at(0), *metric_diagonals);
658 problem.precondition_with_metric(wrap(actions.begin(), actions.begin() + nwork),
659 this->working_set_eigenvalues(), parameters.at(0), metric_actions.at(0));
660 } else if (use_diagonals) {
661 // parameters[0] is overwritten to hand the diagonals to the preconditioner
662 m_handlers->rq().copy(parameters.at(0), *diagonals);
663 problem.precondition(wrap(actions.begin(), actions.begin() + nwork), this->working_set_eigenvalues(),
664 parameters.at(0));
665 } else
666 problem.precondition(wrap(actions.begin(), actions.begin() + nwork), this->working_set_eigenvalues());
667 }
668 nwork = this->end_iteration(parameters, actions);
669 }
670 m_logger->template emit<log::IterationCompleted<scalar_type>>(
671 log::Severity::Normal, log::Verbosity::Info, "Iteration completed ", [&] {
672 return typename log::IterationCompleted<scalar_type>::Payload{iter, roots, {m_errors.begin(), m_errors.end()}, current_values(),
673 std::size_t(std::max(nwork, 0))};
674 });
675 if (verbosity_at_least(Verbosity::Iteration)) {
676 iteration_report();
677 }
678 }
679
680 this->finalize();
681
682 if (verbosity_at_least(Verbosity::Summary)) {
683 summary_report();
684 }
685 // without any errors (e.g. no iteration was done) nothing is known to have converged, and roots without a solution
686 // in the subspace have no error; the non-linear solvers look for a single solution, whatever n_roots() is
687 const bool all_roots = m_errors.size() >= (this->nonlinear() ? 1 : n_roots());
688 const bool errors_converged = all_roots && !m_errors.empty() &&
689 *std::max_element(m_errors.begin(), m_errors.end()) <= m_convergence_threshold;
690 // roots also remain in the working set while their values change by more than the value threshold
691 const bool converged = nwork == 0 and errors_converged;
692 if (verbosity_at_least(Verbosity::Summary)) {
693 if (!all_roots && !m_errors.empty()) {
694 this->m_logger->warn("Subspace holds solutions for only ", m_errors.size(), " of ", n_roots(), " roots");
695 } else if (!errors_converged) {
696 this->m_logger->warn("Solver has not converged to threshold ", m_convergence_threshold);
697 } else if (!converged) {
698 this->m_logger->warn("Solver has not converged to value threshold ", m_convergence_threshold_value);
699 } else {
700 this->m_logger->info("Solver converged");
701 }
702 }
703 m_logger->template emit<log::SolveFinished<scalar_type>>(
704 log::Severity::Normal, log::Verbosity::Debug, "Solve finished ", [&] {
705 return typename log::SolveFinished<scalar_type>::Payload{iter, nwork == 0 and errors_converged,
706 {m_errors.begin(), m_errors.end()},
707 current_values()};
708 });
709 return converged;
710 }

837 bool test_metric(const Problem<R>& problem, R& v0, R& v1, int verbosity, value_type_abs threshold) const {
838 bool success = true;
839 const bool positive = problem.metric_kind() == MetricKind::HermitianPositiveDefinite ||
840 problem.metric_kind() == MetricKind::HermitianPositiveSemidefinite;
841 std::optional<std::pair<Q, Q>> u; // the first test parameter and its action of the metric
842 for (int instance = 0; instance < 2 && problem.test_parameters(instance, v0); ++instance) {
843 problem.metric_action({v0}, {v1});
844 const auto norm = util::norm(m_handlers->rr(), v0);
845 const auto norm_metric_action = util::norm(m_handlers->rr(), v1);
846 const auto vmv = m_handlers->rr().dot(v0, v1);
847 if (std::abs(vmv - conjugate(vmv)) / 2 > threshold * norm * norm_metric_action or
848 (positive and real_part(vmv) < -threshold * norm * norm_metric_action)) {
849 success = false;
850 if (verbosity > -1)
851 std::cout << "<v|M v> = " << vmv << " is not real" << (positive ? " and non-negative" : "") << std::endl;
852 }
853 if (u) {
854 const auto umv = conjugate(m_handlers->rq().dot(v1, u->first));
855 const auto muv = m_handlers->rq().dot(v0, u->second);
856 const auto scale = std::sqrt(std::abs(m_handlers->qq().dot(u->first, u->first)) *
857 std::abs(m_handlers->qq().dot(u->second, u->second))) *
858 norm * norm_metric_action;
859 const auto defect = std::abs(umv - conjugate(muv));
860 if (defect > threshold * std::sqrt(scale)) {
861 success = false;
862 if (verbosity > -1)
863 std::cout << "The metric is not Hermitian: <u|M v> = " << umv << ", <M u|v> = " << conjugate(muv)
864 << std::endl;
865 } else if (verbosity > 0)
866 std::cout << "Hermiticity defect of the metric: " << defect << std::endl;
867 } else
868 u.emplace(m_handlers->qr().copy(v0), m_handlers->qr().copy(v1));
869 }
870 return success;
871 }

1004 prof->start("solve_and_generate_working_set");
1005 // the actions of the metric are not needed any more and serve as scratch space

1007 prof->stop();
1008 read_handler_counts(m_stats, m_handlers);
1009 this->m_end_iteration_needed = true;

1020 size_t solve_and_generate_working_set(const VecRef<R>& parameters, const VecRef<R>& action,
1021 const VecRef<R>& scratch = {}) {
1022 auto prof = profiler();
1023 auto p = prof->push("itsolv::solve_and_generate_working_set");
1024 prof->start("itsolv::solve");
1025 m_subspace_solver->solve(*m_xspace, n_roots());
1026 m_solutions_outdated = false;
1027 prof->stop();
1028 prof->start("itsolv::temp_solutions");
1029 auto nsol = m_subspace_solver->size();
1030 std::vector<std::pair<Q, Q>> temp_solutions{};
1031 const auto batches = detail::parameter_batches(nsol, parameters.size());
1032 // The solutions of a problem with a metric are normalised in the metric, so their Euclidean length, and with it
1033 // the length of the residual, would depend on the scale of the metric. The error is that of the solution of unit
1034 // length instead, which is the same as without a metric.
1035 const auto solution_norms =
1036 m_xspace->metric_kind() != MetricKind::Identity
1037 ? subspace::util::solution_norms(m_subspace_solver->solutions(), m_xspace->data.at(subspace::EqnData::S))
1038 : std::vector<value_type_abs>{};
1039 for (const auto& batch : batches) {
1040 auto [start_sol, end_sol] = batch;
1041 auto roots = std::vector<int>(end_sol - start_sol);
1042 std::iota(begin(roots), end(roots), start_sol);
1043 auto errors = std::vector<value_type_abs>(roots.size(), 0);
1044 solution_with_errors(roots, parameters, action, scratch, solution_norms, errors);
1045 if (batches.size() > 1) {
1046 for (size_t i = 0; i < roots.size(); ++i)
1047 temp_solutions.emplace_back(m_handlers->qr().copy(parameters[i]), m_handlers->qr().copy(action[i]));
1048 m_stats->q_creations += 2 * roots.size();
1049 }
1050 m_subspace_solver->set_error(roots, errors);
1051 }
1052 prof->stop();
1053 combine_errors();
1054 set_value_errors();
1055 m_errors = m_subspace_solver->errors();
1056 m_working_set = detail::select_working_set(parameters.size(), m_errors, m_convergence_threshold, m_value_errors,
1057 m_convergence_threshold_value);
1058 for (size_t i = 0; i < m_working_set.size(); ++i) {

1128 std::vector<P> select_pspace(const Q& diagonals) {
1129 auto selectp = m_handlers->qq().select(m_max_p, diagonals);
1130 if (!selectp.empty()) {
1131 // selectp is keyed by index, not value; find the smallest selected
1132 // diagonal explicitly before applying the threshold.
1133 // the diagonal of a hermitian operator is real, so it is orderable
1134 auto min_val = real_part(std::min_element(selectp.begin(), selectp.end(),
1135 [](const auto& a, const auto& b) {
1136 return real_part(a.second) < real_part(b.second);
1137 })
1138 ->second);
1139 for (auto s = selectp.begin(); s != selectp.end();) {
1140 if (real_part(s->second) > min_val + m_p_threshold)
1141 s = selectp.erase(s);
1142 else
1143 ++s;
1144 }
1145 }
1146 if (!selectp.empty() && selectp.size() < n_roots()) {
1147 this->m_logger->warn("P space smaller than number of roots is not used, starting from the diagonal elements "
1148 "instead; dimension and number of roots",
1149 selectp.size(), n_roots());
1150 selectp.clear();
1151 }
1152 if (!selectp.empty() && verbosity_at_least(Verbosity::Summary)) {
1153 this->m_logger->info("P-space dimension, threshold and limit", selectp.size(), m_p_threshold, m_max_p);
1154 }
1155 if (verbosity_at_least(Verbosity::Detailed)) {
1156 for (const auto& s : selectp) {
1157 this->m_logger->debug(std::format("P space element {}: {:.6e}", s.first, double(real_part(s.second))));
1158 }
1159 }
1160 std::vector<P> pspace;
1161 for (const auto& s : selectp)
1162 pspace.emplace_back((P){{s.first, 1}});
1163 return pspace;
1164 }
