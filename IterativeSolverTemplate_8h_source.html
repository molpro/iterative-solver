26inline std::vector<std::pair<size_t, size_t>> parameter_batches(const size_t nsol, const size_t nparam) {
27 auto batches = std::vector<std::pair<size_t, size_t>>{};
28 if (nparam && nsol) {
29 auto n_batch = nsol / nparam + (nsol % nparam ? 1 : 0);
30 for (size_t ib = 0, start_sol = 0, end_sol = 0; ib < n_batch; ++ib, start_sol = end_sol) {
31 end_sol = std::min(start_sol + nparam, nsol);
32 batches.emplace_back(start_sol, end_sol);
33 }
34 }
35 return batches;
36}

38template <class R, class Q, class P, typename value_type>
39void construct_solution(const VecRef<R>& params, const std::vector<int>& roots,
40 const subspace::Matrix<value_type>& solutions,
41 const std::vector<std::reference_wrapper<P>>& pparams,
42 const std::vector<std::reference_wrapper<Q>>& qparams,
43 const std::vector<std::reference_wrapper<Q>>& dparams, size_t oP, size_t oQ, size_t oD,
44 ArrayHandlers<R, Q, P>& handlers) {
45 if (roots.empty())
46 return;
47 auto prof = molpro::Profiler::single();
48 prof->start("get rd_mat");
49 assert(params.size() >= roots.size());
50 for (size_t i = 0; i < roots.size(); ++i) {
51 handlers.rr().fill(0, params.at(i));
52 }
53 subspace::Matrix<value_type> rp_mat(std::make_pair(pparams.size(), roots.size())),
54 rq_mat(std::make_pair(qparams.size(), roots.size())), rd_mat(std::make_pair(dparams.size(), roots.size()));
55 for (size_t i = 0; i < roots.size(); ++i) {
56 for (size_t j = 0; j < pparams.size(); ++j) {
57 rp_mat(j, i) = solutions(roots[i], oP + j);
58 }
59 for (size_t j = 0; j < qparams.size(); ++j) {
60 rq_mat(j, i) = solutions(roots[i], oQ + j);
61 }
62 for (size_t j = 0; j < dparams.size(); ++j) {
63 rd_mat(j, i) = solutions(roots[i], oD + j);
64 }
65 }
66 prof->stop();
67 handlers.rp().gemm_outer(rp_mat, cwrap(pparams), params);
68 handlers.rq().gemm_outer(rq_mat, cwrap(qparams), params);
69 handlers.rq().gemm_outer(rd_mat, cwrap(dparams), params);
70}

101template <typename T>
102std::vector<int> select_working_set(const size_t nw, const std::vector<T>& errors, const T threshold,
103 const std::vector<T>& value_errors, const T value_threshold) {
104 auto ordered_errors = std::multimap<T, size_t, std::greater<T>>{};
105 for (size_t i = 0; i < errors.size(); ++i) {
106 if (errors[i] > threshold or (i < value_errors.size() and value_errors[i] > value_threshold))
107 ordered_errors.emplace(errors[i], i);
108 }
109 auto working_set = std::vector<int>{};
110 auto end = (ordered_errors.size() < nw ? ordered_errors.end() : next(begin(ordered_errors), nw));
111 std::transform(begin(ordered_errors), end, std::back_inserter(working_set), [](const auto& el) { return el.second; });
112 std::sort(working_set.begin(), working_set.end());
113 return working_set;
114}

505 bool test_problem(const Problem<R>& problem, R& v0, R& v1, int verbosity, value_type_abs threshold) const override {
506 bool success = true;
507 value_type_abs step = 1e-4;
508 if (this->nonlinear()) {
509 if (!problem.test_parameters(0, v0))
510 return true;
511 auto value0 = problem.residual(v0, v1);
512 if (verbosity > 1) {
513 std::cout << "value0 " << value0 << std::endl;
514 // std::cout << "parameters0 " << v0[0] << "," << v0[1] << ",..." << std::endl;
515 // std::cout << "residual0 " << v1[0] << "," << v1[1] << ",..." << std::endl;
516 }
517 Q parameters0 = m_handlers->qr().copy(v0);
518 Q residual0 = m_handlers->qr().copy(v1);
519 for (int instance = 1; problem.test_parameters(instance, v0); ++instance) {
520 m_handlers->rq().axpy(-1.0, parameters0, v0);
521 m_handlers->rr().scal(1 / std::sqrt(std::abs(m_handlers->rr().dot(v0, v0))), v0);
522 Q step1 = m_handlers->qr().copy(v0);
523 auto residual_analytic = m_handlers->rq().dot(v0, residual0);
524 m_handlers->rq().copy(v0, parameters0);
525 m_handlers->rq().axpy(-2*step, step1, v0);
526 auto valuem2 = problem.residual(v0, v1);
527 m_handlers->rq().copy(v0, parameters0);
528 m_handlers->rq().axpy(-1*step, step1, v0);
529 auto valuem1 = problem.residual(v0, v1);
530 m_handlers->rq().copy(v0, parameters0);
531 m_handlers->rq().axpy(+1*step, step1, v0);
532 auto valuep1 = problem.residual(v0, v1);
533 m_handlers->rq().copy(v0, parameters0);
534 m_handlers->rq().axpy(+2*step, step1, v0);
535 auto valuep2 = problem.residual(v0, v1);
536 auto residual_numerical = (valuem2 - value_type(8) * valuem1 + value_type(8) * valuep1 - valuep2) /
537 value_type(12 * step);
538 if (verbosity > 1)
539 std::cout << "testing problem class, instance: " << instance << ", numerical: " << residual_numerical <<", analytical: "<<residual_analytic <<", difference: "<<residual_numerical-residual_analytic << std::endl;
540 success = success && std::abs(residual_numerical - residual_analytic) < threshold;
541 }
542 } else {
543 // checks that the operator applied by apply(x, y) scales linearly; name describes its result
544 auto check_scaling = [&](const auto& apply, const std::string& name) {
545 apply(v0, v1);
546 Q residual = m_handlers->qr().copy(v1);
547 auto norm2_residual = std::sqrt(m_handlers->rr().dot(v1, v1));
548 const value_type_abs scale_factor{10.0};
549 m_handlers->rr().scal(scale_factor, v0);
550 apply(v0, v1);
551 m_handlers->rq().axpy(-scale_factor, residual, v1);
552 auto norm2 = std::sqrt(m_handlers->rr().dot(v1, v1));
553 success = success && std::abs(norm2 / norm2_residual) < threshold;
554 if (verbosity > 0 or (verbosity > -1 and not success))
555 std::cout << "Length of " << name << ": " << norm2_residual << ", scaling defect: " << norm2 << std::endl;
556 };
557 for (int instance = 0; problem.test_parameters(instance, v0); ++instance) {
558 check_scaling([&problem](R& x, R& y) { problem.action({x}, {y}); }, "residual");
559 if (problem.metric_kind() != MetricKind::Identity)
560 check_scaling([&problem](R& x, R& y) { problem.metric_action({x}, {y}); }, "action of the metric");
561 }
562 if (problem.metric_kind() != MetricKind::Identity)
563 success = test_metric(problem, v0, v1, verbosity, threshold) && success;
564 }
565 return success;
566 }

578 bool solve_impl(const VecRef<R>& parameters, const VecRef<R>& actions, const VecRef<R>& metric_actions,
579 const Problem<R>& problem, bool generate_initial_guess) {
580 if (parameters.empty())
581 throw std::runtime_error("Empty container passed to IterativeSolver::solve()");
582 if (parameters.size() != actions.size())
583 throw std::runtime_error("Inconsistent container sizes in IterativeSolver::solve()");
584 // applies the verbosity of the solver to its logger, which repeating does not change
585 if (this->m_verbosity == Verbosity::None) {
586 // Backwards compatibility
587 this->m_logger->set_verbosity(log::Verbosity::None);
588 }
589 if (this->m_verbosity == Verbosity::Detailed) {
590 this->m_logger->set_verbosity(log::Verbosity::Trace);
591 this->m_logger->enable_data_dumps(true);
592 }
593 const auto kind = problem.metric_kind();
594 const bool metric = kind != MetricKind::Identity;
595 const bool store_metric_actions = metric and problem.metric_storage() == MetricStorage::Stored;
596 check_metric_change(kind, store_metric_actions ? MetricStorage::Stored : MetricStorage::OnTheFly);
597 // the problem is checked before the solver changes, using the buffers as scratch space
598 bool use_diagonals = problem.diagonals(actions.at(0));
599 if (generate_initial_guess and not use_diagonals)
600 throw std::runtime_error("Default initial guess requested, but diagonal elements are not available");
601 std::unique_ptr<Q> diagonals;
602 if (use_diagonals)
603 diagonals.reset(new Q{m_handlers->qr().copy(actions.at(0))});
604 std::unique_ptr<Q> metric_diagonals;
605 if (metric and use_diagonals) {
606 if (not problem.metric_diagonals(actions.at(0)))
607 throw std::invalid_argument("The problem provides diagonals() but not metric_diagonals(), without which the "
608 "default preconditioner would ignore the metric");
609 metric_diagonals.reset(new Q{m_handlers->qr().copy(actions.at(0))});
610 }
611 // the initial guess and the P space are selected from the lowest of these values
612 const Q* selection = diagonals.get();
613 std::unique_ptr<Q> ratios;
614 if (metric_diagonals and (generate_initial_guess or m_max_p > 0)) {
615 // the lowest eigenvalues of a generalised eigenproblem are expected at the lowest ratios of the diagonals, which
616 // are exact, as a regulariser would make them depend on the scale of the metric
617 m_handlers->rq().copy(parameters.at(0), *diagonals);
618 detail::precondition_default(wrap_arg(parameters.at(0).get()), std::vector<value_type>{0}, actions.at(0).get(),
619 static_cast<const R*>(nullptr), false);
620 ratios.reset(new Q{m_handlers->qr().copy(parameters.at(0))});
621 selection = ratios.get();
622 }
623 std::vector<P> pspace;
624 if (use_diagonals and m_max_p > 0)
625 pspace = select_pspace(*selection);
626 auto pspace_matrices = pspace_problem_matrices(pspace, problem, metric, store_metric_actions);
627 prepare_solve(problem, actions);
628 if (metric and not store_metric_actions)
629 set_metric(kind, MetricStorage::OnTheFly,
630 [&problem](const CVecRef<R>& parameters, const VecRef<R>& metric_actions) {
631 problem.metric_action(parameters, metric_actions);
632 });
633 else
634 set_metric(kind);
635 int nwork = parameters.size();
636 // a P space replaces any initial guess; without one, max_p > 0 still means the contents of parameters are not used
637 const bool initial_guess =
638 pspace.empty() and (generate_initial_guess or (use_diagonals and m_max_p > 0));
639 if (initial_guess)
640 generate_initial_guess_from_diagonals(parameters, *selection);
641 // the ratios of the diagonal elements are not needed any more
642 selection = nullptr;
643 ratios.reset();
644 if (!pspace.empty())
645 nwork = initialise_pspace(pspace, problem, pspace_matrices, parameters, actions);
646 m_logger->template emit<log::SolveStarted<scalar_type>>(log::Severity::Normal, log::Verbosity::Debug, "Solve started",
647 [&] {
648 return typename log::SolveStarted<scalar_type>::Payload{
649 n_roots(), this->m_max_iter, m_convergence_threshold,
650 pspace.size(), initial_guess};
651 });
652 // the roots worked on in the current iteration, only recorded if the iteration events are wanted
653 std::vector<int> roots;
654 const bool record_roots =
655 m_logger->template is_wanted<log::NewIteration<scalar_type>>(log::Severity::Normal, log::Verbosity::Debug) ||
656 m_logger->template is_wanted<log::IterationCompleted<scalar_type>>(log::Severity::Normal, log::Verbosity::Info);
657 int iter = 0;
658 for (; iter < this->m_max_iter && nwork > 0; iter++) {
659 // the vectors worked on in this iteration: add_vector() takes up no more vectors than there are in the working
660 // set, and in the first iteration after initialising a P space the actions are already known
661 std::size_t naction = 0;
662 if (this->nonlinear())
663 naction = std::min(std::size_t(1), m_working_set.size());
664 else if (iter > 0 or pspace.empty())
665 naction = std::min(size_t(nwork), m_working_set.size());
666 // add_vector() replaces the working set, so it is recorded beforehand
667 if (record_roots)
668 roots.assign(m_working_set.begin(), m_working_set.begin() + naction);
669 m_logger->template emit<log::NewIteration<scalar_type>>(
670 log::Severity::Normal, log::Verbosity::Debug, "Iteration ",
671 [&] { return typename log::NewIteration<scalar_type>::Payload{iter, roots}; });
672 value_type value;
673 if (this->nonlinear()) {
674 value = problem.residual(*parameters.begin(), *actions.begin());
675 nwork = this->add_vector(*parameters.begin(), *actions.begin(), value);
676 } else if (iter > 0 or pspace.empty()) {
677 const auto wparameters = cwrap(parameters.begin(), parameters.begin() + naction);
678 const auto wactions = wrap(actions.begin(), actions.begin() + naction);
679 if (naction > 0 and metric)
680 problem.action_with_metric(wparameters, wactions,
681 wrap(metric_actions.begin(), metric_actions.begin() + naction));
682 else if (naction > 0)
683 problem.action(wparameters, wactions);
684 nwork = metric ? this->add_vector(parameters, actions, metric_actions) : this->add_vector(parameters, actions);
685 }
686 while (this->end_iteration_needed()) {
687 if (nwork > 0) {
688 if (metric_diagonals) {
689 // parameters[0] and metric_actions[0] are overwritten to hand the diagonals to the preconditioner
690 m_handlers->rq().copy(parameters.at(0), *diagonals);
691 m_handlers->rq().copy(metric_actions.at(0), *metric_diagonals);
692 problem.precondition_with_metric(wrap(actions.begin(), actions.begin() + nwork),
693 this->working_set_eigenvalues(), parameters.at(0), metric_actions.at(0));
694 } else if (use_diagonals) {
695 // parameters[0] is overwritten to hand the diagonals to the preconditioner
696 m_handlers->rq().copy(parameters.at(0), *diagonals);
697 problem.precondition(wrap(actions.begin(), actions.begin() + nwork), this->working_set_eigenvalues(),
698 parameters.at(0));
699 } else
700 problem.precondition(wrap(actions.begin(), actions.begin() + nwork), this->working_set_eigenvalues());
701 }
702 nwork = this->end_iteration(parameters, actions);
703 }
704 m_logger->template emit<log::IterationCompleted<scalar_type>>(
705 log::Severity::Normal, log::Verbosity::Info, "Iteration completed ", [&] {
706 return typename log::IterationCompleted<scalar_type>::Payload{iter, roots, {m_errors.begin(), m_errors.end()}, current_values(),
707 std::size_t(std::max(nwork, 0))};
708 });
709 if (verbosity_at_least(Verbosity::Iteration)) {
710 iteration_report();
711 }
712 }
713
714 this->finalize();
715
716 if (verbosity_at_least(Verbosity::Summary)) {
717 summary_report();
718 }
719 // without any errors (e.g. no iteration was done) nothing is known to have converged
720 const bool errors_converged =
721 !m_errors.empty() && *std::max_element(m_errors.begin(), m_errors.end()) <= m_convergence_threshold;
722 // roots also remain in the working set while their values change by more than the value threshold
723 const bool converged = nwork == 0 and errors_converged;
724 if (verbosity_at_least(Verbosity::Summary)) {
725 if (!errors_converged) {
726 this->m_logger->warn("Solver has not converged to threshold ", m_convergence_threshold);
727 } else if (!converged) {
728 this->m_logger->warn("Solver has not converged to value threshold ", m_convergence_threshold_value);
729 } else {
730 this->m_logger->info("Solver converged");
731 }
732 }
733 m_logger->template emit<log::SolveFinished<scalar_type>>(
734 log::Severity::Normal, log::Verbosity::Debug, "Solve finished ", [&] {
735 return typename log::SolveFinished<scalar_type>::Payload{iter, nwork == 0 and errors_converged,
736 {m_errors.begin(), m_errors.end()},
737 current_values()};
738 });
739 return converged;
740 }

747 void solution_impl(const std::vector<int>& roots, const VecRef<R>& parameters, const VecRef<R>& residual,
748 const VecRef<R>& scratch) {
749 // auto prof = profiler()->push("itsolv::solution"); // FIXME two profilers
750 auto prof = molpro::Profiler::single();
751 check_consistent_number_of_roots_and_solutions(roots, parameters.size());
752 prof->start("construct_solution (parameters)");
753 detail::construct_solution(parameters, roots, m_subspace_solver->solutions(), m_xspace->paramsp(),
754 m_xspace->paramsq(), m_xspace->paramsd(), m_xspace->dimensions().oP(),
755 m_xspace->dimensions().oQ(), m_xspace->dimensions().oD(), *m_handlers);
756 prof->stop();
757 auto metric_solutions = detail::MetricSolutions<R, Q>{};
758 auto metric_solutions_q = std::vector<Q>{};
759 if (m_metric_kind != MetricKind::Identity && m_metric_storage == MetricStorage::OnTheFly && !roots.empty()) {
760 prof->start("apply metric");
761 const auto solutions = cwrap(parameters.begin(), parameters.begin() + roots.size());
762 const bool use_scratch = scratch.size() >= roots.size();
763 // without scratch vectors, the residuals hold the actions of the metric until they are copied
764 const auto metric_actions = use_scratch ? wrap(scratch.begin(), scratch.begin() + roots.size())
765 : wrap(residual.begin(), residual.begin() + roots.size());
766 m_apply_metric(solutions, metric_actions);
767 if (use_scratch) {
768 metric_solutions.r = cwrap(metric_actions);
769 } else {
770 for (const auto& m : metric_actions)
771 metric_solutions_q.emplace_back(m_handlers->qr().copy(m));
772 m_stats->q_creations += metric_solutions_q.size();
773 metric_solutions.q = cwrap(metric_solutions_q);
774 }
775 prof->stop();
776 }
777 prof->start("construct_solution (residual)");
778 detail::construct_solution(residual, roots, m_subspace_solver->solutions(), {}, m_xspace->actionsq(),
779 m_xspace->actionsd(), m_xspace->dimensions().oP(), m_xspace->dimensions().oQ(),
780 m_xspace->dimensions().oD(), *m_handlers);
781 prof->stop();
782 prof->start("apply");
783 auto pvectors = detail::construct_vectorP(roots, m_subspace_solver->solutions(), m_xspace->dimensions().oP(),
784 m_xspace->dimensions().nP);
785 if (m_apply_p)
786 m_apply_p(pvectors, m_xspace->cparamsp(), residual);
787 construct_residual(roots, cwrap(parameters), residual, metric_solutions);
788 read_handler_counts(m_stats, m_handlers);
789 prof->stop();
790 }

832 bool test_metric(const Problem<R>& problem, R& v0, R& v1, int verbosity, value_type_abs threshold) const {
833 bool success = true;
834 const bool positive = problem.metric_kind() == MetricKind::HermitianPositiveDefinite ||
835 problem.metric_kind() == MetricKind::HermitianPositiveSemidefinite;
836 std::optional<std::pair<Q, Q>> u; // the first test parameter and its action of the metric
837 for (int instance = 0; instance < 2 && problem.test_parameters(instance, v0); ++instance) {
838 problem.metric_action({v0}, {v1});
839 const auto norm = std::sqrt(std::abs(m_handlers->rr().dot(v0, v0)));
840 const auto norm_metric_action = std::sqrt(std::abs(m_handlers->rr().dot(v1, v1)));
841 const auto vmv = m_handlers->rr().dot(v0, v1);
842 if (std::abs(vmv - conjugate(vmv)) / 2 > threshold * norm * norm_metric_action or
843 (positive and real_part(vmv) < -threshold * norm * norm_metric_action)) {
844 success = false;
845 if (verbosity > -1)
846 std::cout << "<v|M v> = " << vmv << " is not real" << (positive ? " and non-negative" : "") << std::endl;
847 }
848 if (u) {
849 const auto umv = conjugate(m_handlers->rq().dot(v1, u->first));
850 const auto muv = m_handlers->rq().dot(v0, u->second);
851 const auto scale = std::sqrt(std::abs(m_handlers->qq().dot(u->first, u->first)) *
852 std::abs(m_handlers->qq().dot(u->second, u->second))) *
853 norm * norm_metric_action;
854 const auto defect = std::abs(umv - conjugate(muv));
855 if (defect > threshold * std::sqrt(scale)) {
856 success = false;
857 if (verbosity > -1)
858 std::cout << "The metric is not Hermitian: <u|M v> = " << umv << ", <M u|v> = " << conjugate(muv)
859 << std::endl;
860 } else if (verbosity > 0)
861 std::cout << "Hermiticity defect of the metric: " << defect << std::endl;
862 } else
863 u.emplace(m_handlers->qr().copy(v0), m_handlers->qr().copy(v1));
864 }
865 return success;
866 }

1004 size_t solve_and_generate_working_set(const VecRef<R>& parameters, const VecRef<R>& action,
1005 const VecRef<R>& scratch = {}) {
1006 auto prof = profiler();
1007 auto p = prof->push("itsolv::solve_and_generate_working_set");
1008 prof->start("itsolv::solve");
1009 m_subspace_solver->solve(*m_xspace, n_roots());
1010 prof->stop();
1011 prof->start("itsolv::temp_solutions");
1012 auto nsol = m_subspace_solver->size();
1013 std::vector<std::pair<Q, Q>> temp_solutions{};
1014 const auto batches = detail::parameter_batches(nsol, parameters.size());
1015 // The solutions of a problem with a metric are normalised in the metric, so their Euclidean length, and with it
1016 // the length of the residual, would depend on the scale of the metric. The error is that of the solution of unit
1017 // length instead, which is the same as without a metric.
1018 const auto solution_norms =
1019 m_xspace->metric_kind() != MetricKind::Identity
1020 ? subspace::util::solution_norms(m_subspace_solver->solutions(), m_xspace->data.at(subspace::EqnData::S))
1021 : std::vector<value_type_abs>{};
1022 for (const auto& batch : batches) {
1023 auto [start_sol, end_sol] = batch;
1024 auto roots = std::vector<int>(end_sol - start_sol);
1025 std::iota(begin(roots), end(roots), start_sol);
1026 solution_impl(roots, parameters, action, scratch);
1027 auto errors = std::vector<value_type_abs>(roots.size(), 0);
1028 compute_errors(roots, cwrap(parameters), cwrap(action), solution_norms, errors);
1029 if (batches.size() > 1) {
1030 for (size_t i = 0; i < roots.size(); ++i)
1031 temp_solutions.emplace_back(m_handlers->qr().copy(parameters[i]), m_handlers->qr().copy(action[i]));
1032 m_stats->q_creations += 2 * roots.size();
1033 }
1034 m_subspace_solver->set_error(roots, errors);
1035 }
1036 prof->stop();
1037 combine_errors();
1038 set_value_errors();

1108 std::vector<P> select_pspace(const Q& diagonals) {
1109 auto selectp = m_handlers->qq().select(m_max_p, diagonals);
1110 if (!selectp.empty()) {
1111 // selectp is keyed by index, not value; find the smallest selected
1112 // diagonal explicitly before applying the threshold.
1113 // the diagonal of a hermitian operator is real, so it is orderable
1114 auto min_val = real_part(std::min_element(selectp.begin(), selectp.end(),
1115 [](const auto& a, const auto& b) {
1116 return real_part(a.second) < real_part(b.second);
1117 })
1118 ->second);
1119 for (auto s = selectp.begin(); s != selectp.end();) {
1120 if (real_part(s->second) > min_val + m_p_threshold)
1121 s = selectp.erase(s);
1122 else
1123 ++s;
1124 }
1125 }
1126 if (!selectp.empty() && selectp.size() < n_roots()) {
1127 this->m_logger->warn("P space smaller than number of roots is not used, starting from the diagonal elements "
1128 "instead; dimension and number of roots",
1129 selectp.size(), n_roots());
1130 selectp.clear();
1131 }
1132 if (!selectp.empty() && verbosity_at_least(Verbosity::Summary)) {
1133 this->m_logger->info("P-space dimension, threshold and limit", selectp.size(), m_p_threshold, m_max_p);
1134 }
1135 if (verbosity_at_least(Verbosity::Detailed)) {
1136 for (const auto& s : selectp) {
1137 this->m_logger->debug(std::format("P space element {}: {:.6e}", s.first, double(real_part(s.second))));
1138 }
1139 }
1140 std::vector<P> pspace;
1141 for (const auto& s : selectp)
1142 pspace.emplace_back((P){{s.first, 1}});
1143 return pspace;
