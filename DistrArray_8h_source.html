30 template<typename Array>
31 class ValueProxy {
32 public:
33 ValueProxy(const ValueProxy &) = delete;
34 ValueProxy(ValueProxy &&) = delete;
35
36 // Assigning a proxy writes the *value* it refers to into this proxy's cell,
37 // rather than rebinding this proxy to a different array/index -- that keeps
38 // *it1 = *it2 (as used by std::copy et al.) working between two DistrArrays.
39 ValueProxy &operator=(const ValueProxy &other) { return *this = static_cast<typename Array::value_type>(other); }
40 ValueProxy &operator=(ValueProxy &&other) { return *this = static_cast<typename Array::value_type>(other); }
41 template <typename OtherArray>
42 ValueProxy &operator=(const ValueProxy<OtherArray> &other) {
43 return *this = static_cast<typename OtherArray::value_type>(other);
44 }
45
46 ValueProxy &operator=(Array::value_type val) {
47 m_arr.set(m_idx, std::move(val));
48
49 return *this;
50 }
51
52 operator typename Array::value_type() const {
53 return m_arr.at(m_idx);
54 }
55
56 private:
57 std::add_lvalue_reference_t<Array> m_arr;
58 Array::index_type m_idx;
59
60 friend std::remove_const_t<Array>;
61
62 ValueProxy(std::add_lvalue_reference_t<Array> arr, Array::index_type idx) : m_arr(arr), m_idx(idx) {}
63 };

364template<bool is_const>
365class DistrArrayIteratorImpl {
366 public:
367 using value_type = std::conditional_t<
368 is_const, DistrArray::value_type, ValueProxy<DistrArray>>;
369 using reference = value_type;
370 using const_reference = reference;
371 using difference_type = std::ptrdiff_t;
372 // *this dereferences to a proxy, not an lvalue reference, so there is no meaningful
373 // pointer type. iterator_category is spelled out explicitly because std::iterator_traits'
374 // legacy (pre-C++20) fallback cannot deduce past input_iterator_tag for a proxy iterator,
375 // which would make std::fill/std::copy etc. reject the single-pass-only overloads even
376 // though this iterator satisfies std::random_access_iterator (see the static_assert below).
377 using pointer = void;
378 using iterator_category = std::random_access_iterator_tag;
379
380 DistrArrayIteratorImpl() = default;
381 DistrArrayIteratorImpl(std::conditional_t<is_const, const DistrArray, DistrArray> &array,
382 DistrArray::index_type pos) : m_arr(&array), m_pos(pos) {}
383
384 DistrArrayIteratorImpl &operator+=(difference_type val) {
385 m_pos += val;
386 return *this;
387 }
388 friend DistrArrayIteratorImpl operator+(const DistrArrayIteratorImpl &it, difference_type val) {
389 return DistrArrayIteratorImpl(*it.m_arr, it.m_pos + val);
390 }
391 friend DistrArrayIteratorImpl operator+(difference_type val, const DistrArrayIteratorImpl &it) {
392 return it + val;
393 }
394 DistrArrayIteratorImpl &operator++() {
395 ++m_pos;
396 return *this;
397 }
398 DistrArrayIteratorImpl operator++(int) {
399 DistrArrayIteratorImpl copy(*this);
400 ++m_pos;
401 return copy;
402 }
403
404 DistrArrayIteratorImpl &operator-=(difference_type val) {
405 m_pos -= val;
406 return *this;
407 }
408 friend DistrArrayIteratorImpl operator-(const DistrArrayIteratorImpl &it, difference_type val) {
409 return DistrArrayIteratorImpl(*it.m_arr, it.m_pos - val);
410 }
411 DistrArrayIteratorImpl &operator--() {
412 --m_pos;
413 return *this;
414 }
415 DistrArrayIteratorImpl operator--(int) {
416 DistrArrayIteratorImpl copy(*this);
417 --m_pos;
418 return copy;
419 }
420
421 reference operator*() const { return (*m_arr)[m_pos]; }
422
423 reference operator[](difference_type offset) const { return (*m_arr)[m_pos + offset]; }
424
425 difference_type operator-(const DistrArrayIteratorImpl &other) const { return m_pos - other.m_pos; }
426
427 bool operator==(const DistrArrayIteratorImpl &) const = default;
428 std::strong_ordering operator<=>(const DistrArrayIteratorImpl &) const = default;
429
430 private:
431 std::conditional_t<is_const, const DistrArray, DistrArray> *m_arr = nullptr;
432 DistrArray::index_type m_pos = 0;
433};
