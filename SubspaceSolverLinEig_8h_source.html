28 m_logger->trace("SubspaceSolverLinEig::solve");
29 if (xspace.data.find(EqnData::rhs) == xspace.data.end() || xspace.data[EqnData::rhs].empty()) {
30 solve_eigenvalue(xspace, nroots_max);

52 void solve_eigenvalue(IXSpace<R, Q, P>& xspace, const size_t nroots_max) {
53 m_logger->trace("SubspaceSolverLinEig::solve_eigenvalue");
54 auto h = xspace.data[EqnData::H];
55 auto s = xspace.data[EqnData::S];
56 m_logger->data_dump("S = ", s);
57 m_logger->data_dump<15>("H = ", h);
58 auto dim = h.rows();
59 auto evec = std::vector<value_type>{};
60 int verbosity = convert_verbosity(m_logger->verbosity());
61 itsolv::eigenproblem(evec, m_eigenvalues, h.data(), s.data(), dim, m_hermitian, m_svd_solver_threshold, verbosity,
62 &m_imag_eigval_comps);
63 size_t n_solutions = 0;
64 if (dim)
65 n_solutions = evec.size() / dim;
66 auto full_matrix = Matrix<value_type>{std::move(evec), {n_solutions, dim}};
67 auto nroots = std::min(nroots_max, n_solutions);
68 m_eigenvalues.resize(nroots);
69 auto [first, last] = std::ranges::remove_if(
70 m_imag_eigval_comps, [nroots](std::size_t idx) { return idx >= nroots; },
71 &decltype(m_imag_eigval_comps)::value_type::first);
72 if (first != last && (std::ranges::distance(first, last) % 2) != 0) {
73 m_logger->info("Complex eigenvalue pair split up due to truncation to requested number of roots");
74 }
75 m_imag_eigval_comps.erase(first, last);
76 m_solutions.resize({nroots, dim});
77 m_solutions.slice() = full_matrix.slice({0, 0}, {nroots, dim});
78 m_errors.assign(size(), std::numeric_limits<value_type_abs>::max());
79 m_logger->data_dump<10>("eigenvalues = ", m_eigenvalues);
80 if (!m_imag_eigval_comps.empty()) {
81 m_logger->info("The following eigenvalues turned out to be complex-valued: ",
82 m_imag_eigval_comps |
83 std::ranges::views::transform([](const auto& pair) { return pair.first + 1; }));
84 m_logger->data_dump("imaginary parts of eigenvalues = ",
85 m_imag_eigval_comps | std::ranges::views::transform([](auto pair) {
86 pair.first += 1;
87 return pair;
88 }));
89 }
90 m_logger->data_dump("eigenvectors = ", m_solutions);
91 }

93 void solve_linear_equations(IXSpace<R, Q, P>& xspace) {
94 m_logger->trace("SubspaceSolverLinEig::solve_linear_equations");
95 auto h = xspace.data[EqnData::H];
96 auto s = xspace.data[EqnData::S];
97 auto rhs = xspace.data[EqnData::rhs];
98 m_logger->data_dump<15>("S = ", s);
99 m_logger->data_dump<15>("H = ", h);
100 m_logger->data_dump<15>("rhs = ", rhs);
101 const auto dim = h.rows();
102 const auto n_solutions = rhs.cols();
103 auto solution = std::vector<value_type>{};
104 m_eigenvalues.assign(n_solutions, 0);
105 m_imag_eigval_comps.clear();
106 int verbosity = convert_verbosity(m_logger->verbosity());
107 itsolv::solve_LinearEquations(solution, m_eigenvalues, h.data(), s.data(), rhs.data(), dim, n_solutions,
108 m_augmented_hessian, m_svd_solver_threshold, verbosity);
109 m_solutions = Matrix<value_type>{std::move(solution), {n_solutions, dim}};
110 m_errors.assign(size(), std::numeric_limits<value_type_abs>::max());
111 m_logger->data_dump<10>("eigenvalues = ", m_eigenvalues);
112 m_logger->data_dump("solutions = ", m_solutions);
113 }
