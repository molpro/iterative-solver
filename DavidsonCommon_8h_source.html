66 size_t end_iteration(const VecRef<R>& parameters, const VecRef<R>& action) override {
67 auto prof = this->profiler();
68 auto m_prof = prof->push("itsolv::end_iteration");
69 if (this->m_working_set.empty() && !m_dspace_resetter.in_progress()) {
70 // all roots have converged: the subspace is left as it is
71 m_resetting_in_progress = false;
72 } else if (m_dspace_resetter.do_reset(this->m_stats->iterations, this->m_xspace->dimensions())) {
73 m_resetting_in_progress = true;
74 // retained_solutions() can hold a truncated complex pair partner beyond the roots; with no more parameters than
75 // roots, it is handed out under the index of a root
76 const auto nparam = std::min(parameters.size(), this->n_roots());
77 this->m_working_set =
78 m_dspace_resetter.run(VecRef<R>(parameters.begin(), parameters.begin() + nparam), *this->m_xspace,
79 this->m_subspace_solver->retained_solutions(), rspace_opts.norm_thresh,
80 rspace_opts.svd_thresh, *this->m_handlers, *this->m_logger);
81 this->m_solutions_outdated = true;
82 } else {
83 m_resetting_in_progress = false;
84 prof->start("end_iteration (propose_rspace)");
85 this->m_working_set =
86 detail::propose_rspace(*this, parameters, action, *this->m_xspace, *this->m_subspace_solver,
87 *this->m_handlers, *this->m_logger, rspace_opts, qspace_opts, *this->profiler().get());
88 prof->stop();
89 }
90 this->m_stats->iterations++;
91 read_handler_counts(this->m_stats, this->m_handlers);
92 this->m_end_iteration_needed = false;
93 return this->working_set().size();
94 }

155 void set_davidson_options(const Options& options, const DavidsonOptions& opt) {

170 if (opt.norm_thresh)

210 void set_value_errors() override {
211 const auto current_values = values();
212 this->m_value_errors.assign(current_values.size(), std::numeric_limits<value_type_abs>::max());
213 for (size_t i = 0; i < std::min(m_last_values.size(), current_values.size()); i++)
214 this->m_value_errors[i] = std::abs(current_values[i] - m_last_values[i]);
215 // resetting the D space does not change the solutions, so the change measured across it would be meaningless
216 if (!m_resetting_in_progress)
217 m_last_values = current_values;
218 }
